2"""Mémoire vectorielle Qdrant (mode embarqué par défaut : aucun serveur, aucune clé).
3
4Deux collections :
5 veille_passages : textes sources verbatim (unité de preuve) + métadonnées filtrables (pays, produit, source)
6 veille_records : enregistrements canoniques validés (Pydantic), interrogeables par similarité + filtres
7"""
8from __future__ import annotations
9import hashlib, json, sys, uuid, warnings
10warnings.filterwarnings("ignore", message="Api key is used with an insecure connection")
11from dataclasses import asdict
12from qdrant_client import QdrantClient, models as qm
13from . import config, llm
14from .sources import Passage

16PASSAGES, RECORDS = "veille_passages", "veille_records"
17_client = None

20def client() -> QdrantClient:
21 global _client
22 if _client is None:
23 if not config.QDRANT_FORCE_EMBEDDED:
24 try:
25 c = QdrantClient(url=config.QDRANT_URL, api_key=config.QDRANT_API_KEY, check_compatibility=False, timeout=10)
26 c.get_collections() # sonde : serveur joignable et clé acceptée ?
27 _client = c
28 except Exception as e:
29 print(f"[store] Qdrant serveur indisponible ({type(e).__name__}) -> mode embarqué", file=sys.stderr)
30 if _client is None:
31 config.QDRANT_PATH.mkdir(parents=True, exist_ok=True)
32 _client = QdrantClient(path=str(config.QDRANT_PATH))
33 return _client

36def pid(key: str) -> str:
37 return str(uuid.uuid5(uuid.NAMESPACE_URL, key))

40def cached_embed(text: str) -> list[float]:
41 p = config.CACHE / f"emb_{hashlib.sha256((config.EMBED_MODEL + text).encode()).hexdigest()}.json"
42 if p.is_file():
43 return json.loads(p.read_text())
44 v = llm.embed(text)
45 p.write_text(json.dumps(v))
46 return v

49def _ensure(name: str, size: int):
50 c = client()
51 if not c.collection_exists(name):
52 c.create_collection(name, vectors_config=qm.VectorParams(size=size, distance=qm.Distance.COSINE))

55def _flt(**kw):
56 must = [qm.FieldCondition(key=k, match=qm.MatchValue(value=v)) for k, v in kw.items() if v]
57 return qm.Filter(must=must) if must else None

60def reset():
61 """Repart d'un index propre (collections lpc_* uniquement ; les autres collections ne sont jamais touchées)."""
62 for name in (PASSAGES, RECORDS):
63 if client().collection_exists(name):
64 client().delete_collection(name)
65
66
67def index_passages(passages: list[Passage]) -> int:
68 pts = []
69 for p in passages:
70 v = cached_embed(p.text)
71 _ensure(PASSAGES, len(v))
72 payload = asdict(p)
73 pts.append(qm.PointStruct(id=pid(p.passage_id), vector=v, payload=payload))
74 if pts:
75 client().upsert(PASSAGES, pts)
76 return len(pts)

79def get_passage(passage_id: str) -> Passage | None:
80 """Retrouve un passage par son identifiant (relecture humaine : on repart du texte source d'origine)."""
81 if not client().collection_exists(PASSAGES):
82 return None
83 points, _ = client().scroll(PASSAGES, scroll_filter=_flt(passage_id=passage_id), limit=1, with_payload=True)
84 return Passage(**points[0].payload) if points else None
85
86
87def search_passages(query: str, k: int = 5, **filters) -> list[dict]:
88 v = cached_embed(query)
89 if not client().collection_exists(PASSAGES):
90 return []
91 hits = client().query_points(PASSAGES, query=v, limit=k, query_filter=_flt(**filters), with_payload=True).points
92 return [{"score": round(h.score, 4), **h.payload} for h in hits]

94
95def all_passages(**filters) -> list[Passage]:
96 if not client().collection_exists(PASSAGES):
97 return []
98 out, off = [], None
99 while True:
100 pts, off = client().scroll(PASSAGES, scroll_filter=_flt(**filters), limit=256, offset=off, with_payload=True)
101 out += [Passage(**p.payload) for p in pts]
102 if off is None:
103 break
104 return sorted(out, key=lambda p: p.passage_id)

107def record_text(r: dict) -> str:
108 return (f"{r['product']} {r['country']} culture {r.get('crop_original') or ''} cible {r.get('target_original') or ''} "
109 f"dose {r.get('dose_original') or 'non indiquée'} statut {r.get('authorization_status') or ''}")

112def upsert_records(records: list[dict]) -> int:
113 pts = []
114 for r in records:
115 v = cached_embed(record_text(r))
116 _ensure(RECORDS, len(v))
117 pts.append(qm.PointStruct(id=pid(r["record_id"]), vector=v, payload=r))
118 if pts:
119 client().upsert(RECORDS, pts)
120 return len(pts)
