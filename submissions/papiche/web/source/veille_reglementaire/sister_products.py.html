2"""Détection de produits sœurs : mêmes souche et substance active sous des noms commerciaux différents.
3
4Score DÉTERMINISTE et lisible (somme pondérée de critères observables) ; la justification énumère les critères
5vérifiés. Une correspondance reste toujours « a_confirmer » : seule l'équipe Regulatory peut la valider.
6"""
7from __future__ import annotations
8import json, re
9from itertools import combinations
10from . import config
11from .eppo_codes import fold
12from .models import SisterHypothesis

14W = {"meme_souche": 0.5, "meme_espece": 0.2, "meme_titulaire": 0.1, "meme_formulation": 0.1,
15 "meme_famille_marque": 0.05, "noms_proches": 0.05}
16STRAIN = re.compile(r"(?:souche|strain|stam)\s*:?\s*([A-Za-z0-9][A-Za-z0-9/\-. ]{1,18}?)(?=[\s(),;]|$)", re.I)
17FORM = re.compile(r"\b(WG|WP|OD|GR|SC|EC|WS|SL|DP)\b")
18
19
20def canon(name: str) -> str:
21 """'BOTANIGARD 22WP' == 'Botanigard 22 WP' == 'PRESTOP (WG)' == 'PRESTOP WG'."""
22 return re.sub(r"[^a-z0-9]", "", fold(name))
23
24
25def _syn() -> dict:
26 d = json.loads((config.CONFIG / "taxon_synonyms.json").read_text())
27 return {k: v for k, v in d.items() if not k.startswith("_")}

30def profile(product: str, ai: str, holder: str) -> dict:
31 a = fold(ai)
32 species = None
33 m = re.match(r"([a-z]+ [a-z]+)", a)
34 if m:
35 species = m.group(1)
36 if species.split(" ")[1] in ("souche", "strain", "stam", "k61"): # espèce non nommée : genre seul
37 species = species.split(" ")[0] + " sp"
38 species = _syn().get(species, species)
39 strain = None
40 for m in STRAIN.finditer(ai or ""):
41 strain = fold(m.group(1)).replace(" ", "")
42 if strain:
43 break
44 form = FORM.search(product.upper())
45 return {"species": species, "strain": strain, "formulation": form.group(1) if form else None,
46 "holder": fold(holder), "brand": fold(product).split(" ")[0]}

49def detect(passages_meta: list[dict], threshold: float = 0.55) -> list[dict]:
50 """passages_meta : [{product, country, ai, holder}] dédoublonnés par (product, country)."""
51 uniq = {}
52 for p in passages_meta:
53 uniq.setdefault((p["product"], p["country"]), p)
54 profs = {k: profile(v["product"], v["ai"], v["holder"]) for k, v in uniq.items()}
55 out = []
56 for (ka, pa), (kb, pb) in combinations(profs.items(), 2):
57 if canon(ka[0]) == canon(kb[0]):
58 continue # même produit (orthographe près) : pas une hypothèse
59 f = {
60 "meme_souche": bool(pa["strain"] and pa["strain"] == pb["strain"]),
61 "meme_espece": bool(pa["species"] and pa["species"] == pb["species"]),
62 "meme_titulaire": bool(pa["holder"] and pa["holder"].split(" ")[0] == pb["holder"].split(" ")[0]),
63 "meme_formulation": bool(pa["formulation"] and pa["formulation"] == pb["formulation"]),
64 "meme_famille_marque": pa["brand"] == pb["brand"],
65 "noms_proches": fold(ka[0]) in fold(kb[0]) or fold(kb[0]) in fold(ka[0]),
66 }
67 if not f["meme_espece"]:
68 continue
69 score = round(sum(W[k] for k, v in f.items() if v), 2)
70 if score < threshold:
71 continue
72 why = [k.replace("_", " ") for k, v in f.items() if v]
73 out.append(SisterHypothesis(
74 product_a=ka[0], country_a=ka[1], product_b=kb[0], country_b=kb[1], confidence=score, features=f,
75 justification=(f"{ka[0]} ({ka[1]}) et {kb[0]} ({kb[1]}) : critères vérifiés dans les sources — "
76 + ", ".join(why) + f". Souche lue : {pa['strain'] or 'n/a'} / {pb['strain'] or 'n/a'}. "
77 "Hypothèse à confirmer par Regulatory (composition et formulation exactes non comparées).")
78 ).model_dump())
79 return sorted(out, key=lambda x: -x["confidence"])
