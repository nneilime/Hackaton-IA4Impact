22def french(number: float, decimals: int = 0) -> str:
23 """Nombre au format français, lisible à voix haute : 117,7 ; 16,26 ; 400 000."""
24 text = f"{number:,.{decimals}f}".replace(",", " ").replace(".", ",")
25 return text

28def collect_figures() -> dict:
29 """Tous les chiffres que la voix peut citer, relus dans les résultats réels."""
30 from . import web_data # import tardif : web_data utilise ce module
31 traces = pipeline.load("traces.jsonl")
32 performance = web_data.measure_performance(traces)
33 history = web_data.read_prompt_history()
34 network = web_data.read_network_runs()
35 evaluation = json.loads((config.PROCESSED / "eval.json").read_text())
36 bench = json.loads((config.PROCESSED / "bench" / "report.json").read_text())
37 cost = json.loads((config.PROCESSED / "cost" / "report.json").read_text())
38
39 holdout_path = config.PROCESSED / "holdout.json"
40 holdout = json.loads(holdout_path.read_text()) if holdout_path.exists() else None
41 attempts = sum(int(k) * v for k, v in evaluation["essais_par_passage"].items())
42 big = cost["scenarios"][2]
43 api_options = [o for o in cost["options"] if o["kind"] != "local"]
44 flagship = next(o for o in cost["options"] if o["kind"] == "api" and o.get("tier") == "haut" and "Opus" in o["label"])
45 recommended = bench["recommendation"]
46 return {
47 "passages": performance["passages"], "attempts": attempts, "refused": attempts - performance["published"],
48 "first_pct": round(100 * performance["first_try"] / performance["passages"]),
49 "published_pct": round(100 * performance["published_rate"]), "rescued": performance["rescued_by_retry"],
50 "pydantic_refused": performance["refused_by_pydantic"], "proof_refused": performance["refused_by_proof_check"],
51 "dose_pct": round(100 * evaluation["exactitude_par_champ"]["dose_value"]),
52 "v1_expert": history[0]["sent_to_expert"], "v3_expert": history[-1]["sent_to_expert"],
53 "holdout_passages": holdout["passages"] if holdout else 0,
54 "holdout_dose_pct": round(100 * holdout["exactitude_par_champ"]["dose_value"]) if holdout else 0,
55 "requests": network["cold"]["total_requests"], "cold_mb": french(network["cold"]["total_megabytes"], 1),
56 "warm_mb": french(network["warm"]["total_megabytes"], 1) if network["warm"] else "0",
57 "vram": french(recommended["vram_1_request_gb"], 1), "passages_big": french(big["passages"]),
58 "local_cost": french(cost["options"][0]["totals"][big["name"]], 0),
59 "cheapest_cost": french(min(o["totals"][big["name"]] for o in api_options), 0),
60 "flagship_cost": french(flagship["totals"][big["name"]], 0),
61 }

64def render_script() -> list[str]:
65 """Le texte de chaque diapositive, chiffres remplacés."""
66 templates = json.loads((config.CONFIG / "narration_fr.json").read_text(encoding="utf-8"))["slides"]
67 figures = collect_figures()
68 return [template.format(**figures) for template in templates]

73def synthesize(text: str, engine: str, voice: str, wav_path) -> None:
74 if engine == "orpheus":
75 response = requests.post(config.TTS_URL or DEFAULT_TTS_URL, timeout=600,
76 json={"model": "orpheus", "input": text, "voice": voice, "response_format": "wav"})
77 response.raise_for_status()
78 wav_path.write_bytes(response.content)
79 elif engine == "espeak": # voix de secours, hors ligne, mais robotique
80 subprocess.run(["espeak", "-v", "fr", "-s", "150", "-w", str(wav_path), text], check=True)
81 else:
82 raise SystemExit(f"Moteur inconnu : {engine} (orpheus ou espeak)")

85def audio_duration_seconds(path) -> float:
86 out = subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", str(path)],
87 capture_output=True, text=True, check=True)
88 return float(out.stdout.strip())

91def convert_to_mp3(wav_path, mp3_path) -> float:
92 """Convertit en mp3 léger et accélère légèrement si la phrase dépasse 20 s. Retourne la durée finale."""
93 duration = audio_duration_seconds(wav_path)
94 tempo = min(1.5, max(1.0, duration / MAX_SECONDS_PER_SLIDE))
95 subprocess.run(["ffmpeg", "-y", "-loglevel", "error", "-i", str(wav_path), "-filter:a", f"atempo={tempo:.3f}",
96 "-ac", "1", "-ar", "24000", "-b:a", "64k", str(mp3_path)], check=True)
97 return audio_duration_seconds(mp3_path)

100def render_audio(engine: str = "orpheus", voice: str = "amelie", only: list[int] | None = None) -> list[dict]:
101 """Génère les fichiers audio ; `only` = numéros de diapositives à refaire (toutes par défaut)."""
102 AUDIO_FOLDER.mkdir(parents=True, exist_ok=True)
103 report = []
104 for number, text in enumerate(render_script(), 1):
105 if only and number not in only:
106 continue
107 wav_path, mp3_path = AUDIO_FOLDER / f"pk-{number:02d}.wav", AUDIO_FOLDER / f"pk-{number:02d}.mp3"
108 synthesize(text, engine, voice, wav_path)
109 seconds = convert_to_mp3(wav_path, mp3_path)
110 wav_path.unlink()
111 report.append({"slide": number, "seconds": round(seconds, 1), "fits_20s": seconds <= 20.0})
112 print(f" diapositive {number:02d} : {seconds:.1f} s", flush=True)
113 return report
