2"""Estimation du coût : service d'IA local (RTX 3090) comparé aux API payantes au token et aux forfaits.
3
4Les volumes de tokens viennent du benchmark (tokens réellement consommés par passage, essais et contrôles inclus).
5Les tarifs viennent de config/pricing.json (relevés sur les pages officielles, avec source et date).
6Les hypothèses locales (prix de la carte, électricité…) sont dans config/local_cost_assumptions.json : à ajuster.
7Tout est exprimé en dollars US, comme les tarifs publics.
8"""

15SCENARIOS = [
16 ("Démonstration (corpus actuel)", None), # nombre de passages lu dans le benchmark
17 ("Mise à jour quotidienne pendant 1 an (hypothèse : 500 passages par jour)", 500 * 365),
18 ("Chargement initial FR + CA + US", 400_000),
19 ("Monde (≈ 10 pays)", 1_000_000),
20]

23def load_json(path) -> dict:
24 return json.loads(path.read_text(encoding="utf-8"))
25
26
27def local_cost_per_passage(seconds_per_passage: float, assumptions: dict) -> dict:
28 """Coût d'un passage sur la carte locale : électricité (coût marginal) et amortissement de la carte."""
29 kilowatt_hours = assumptions["gpu_power_watts"] / 1000 * seconds_per_passage / 3600
30 electricity = kilowatt_hours * assumptions["electricity_usd_per_kwh"]
31 busy_hours_over_lifetime = assumptions["lifetime_years"] * 8760 * assumptions["utilization"]
32 hardware = assumptions["gpu_purchase_usd"] / busy_hours_over_lifetime * seconds_per_passage / 3600
33 return {"electricity": electricity, "hardware": hardware, "kwh": kilowatt_hours}

35
36def api_cost_per_passage(model: dict, prompt_tokens: float, generated_tokens: float, batch_discount: float = 1.0) -> float:
37 cost = prompt_tokens * model["input"] / 1e6 + generated_tokens * model["output"] / 1e6
38 return cost * batch_discount

41def build_report(bench_report: dict) -> dict:
42 pricing = load_json(config.CONFIG / "pricing.json")
43 assumptions = load_json(config.CONFIG / "local_cost_assumptions.json")
44 chosen = bench_report["recommendation"]["model"]
45 measured = bench_report["results"][chosen]
46 prompt_tokens, generated_tokens = measured["prompt_tokens_per_passage"], measured["gen_tokens_per_passage"]
47 demo_passages = bench_report["storage"]["passages_now"]
48 scenarios = [(name, demo_passages if n is None else n) for name, n in SCENARIOS]
49
50 local = local_cost_per_passage(measured["s_per_passage_mean"], assumptions)
51 options = [
52 {"kind": "local", "label": f"Service local : RTX 3090, {chosen} (électricité seule)", "per_passage": local["electricity"]},
53 {"kind": "local", "label": f"Service local : RTX 3090, {chosen} (électricité + amortissement)",
54 "per_passage": local["electricity"] + local["hardware"]},
55 ]
56 for model in pricing["api_models"]:
57 full_name = model["model"] if model["model"].startswith(model["vendor"]) else f"{model['vendor']} {model['model']}"
58 options.append({"kind": "api", "label": full_name, "tier": model["tier"], "note": model.get("note"),
59 "source": model["source"], "price_in": model["input"], "price_out": model["output"],
60 "per_passage": api_cost_per_passage(model, prompt_tokens, generated_tokens)})
61 options.append({"kind": "api_batch", "label": f"{full_name} (Batch, −50 %)", "tier": model["tier"],
62 "source": model["source"], "per_passage": api_cost_per_passage(model, prompt_tokens, generated_tokens, pricing["batch_discount"])})
63 for option in options:
64 option["totals"] = {name: round(option["per_passage"] * count, 2) for name, count in scenarios}
65 option["per_1000_passages"] = round(option["per_passage"] * 1000, 4)
66
67 return {
68 "checked_on": pricing["checked_on"], "model_measured": chosen,
69 "tokens_per_passage": {"prompt": prompt_tokens, "generated": generated_tokens},
70 "seconds_per_passage": measured["s_per_passage_mean"], "scenarios": [{"name": n, "passages": c} for n, c in scenarios],
71 "local_assumptions": assumptions, "local_per_passage": local, "options": options,
72 "subscriptions": pricing["subscriptions"], "subscription_facts": pricing["subscription_facts"],
73 "caveats": [
74 "Les tokens par passage sont ceux du modèle local mesuré (tokenizer différent chez chaque fournisseur : écart possible de quelques dizaines de %).",
75 "Seuls les modèles locaux ont été mesurés en qualité : le coût d'une API ne dit rien de sa qualité sur cette tâche.",
76 "Les coûts locaux supposent le service déjà en place ; hors maintenance, réseau et temps d'exploitation.",
77 "Le prix d'achat de la carte, l'électricité et le taux d'utilisation sont des hypothèses (config/local_cost_assumptions.json).",
78 ],
79 }

82def format_money(value: float) -> str:
83 """Format français : 1 030,76 $ (espace insécable entre milliers)."""
84 if value < 0.01:
85 return "< 0,01 $"
86 return f"{value:,.2f}".replace(",", "\u202f").replace(".", ",") + " $"
87
88
89def format_count(number: int) -> str:
90 return f"{number:,}".replace(",", "\u202f")

93def render_markdown(report: dict) -> str:
94 names = [s["name"] for s in report["scenarios"]]
95 lines = [
96 "# Coût : service d'IA local, API au token, forfaits", "",
97 f"Tarifs relevés le {report['checked_on']} (sources dans `config/pricing.json`). Montants en dollars US, hors taxes.",
98 f"Volume mesuré sur `{report['model_measured']}` : **{report['tokens_per_passage']['prompt']} tokens lus + "
99 f"{report['tokens_per_passage']['generated']} tokens écrits par passage** (essais et contrôles inclus), "
100 f"{report['seconds_per_passage']} s par passage sur la RTX 3090.", "",
101 "## Coût total par scénario", "",
102 "| Option | " + " | ".join(f"{s['name']} ({format_count(s['passages'])} passages)" for s in report["scenarios"]) + " |",
103 "|---|" + "---|" * len(names)]
104 for option in report["options"]:
105 lines.append(f"| {option['label']} | " + " | ".join(format_money(option["totals"][n]) for n in names) + " |")
106 a = report["local_assumptions"]
107 lines += ["", "## Hypothèses du coût local", "",
108 f"Carte {a['gpu_purchase_usd']} $ amortie sur {a['lifetime_years']} ans à {int(a['utilization'] * 100)} % d'utilisation ; "
109 f"{a['gpu_power_watts']} W ; électricité {a['electricity_usd_per_kwh']} $/kWh. "
110 f"Énergie par passage : {report['local_per_passage']['kwh'] * 1000:.2f} Wh.", "",
111 "## Forfaits (abonnements par personne)", "", "| Fournisseur | Forfait | Prix mensuel | Remarque |", "|---|---|---|---|"]
112 lines += [f"| {s['vendor']} | {s['plan']} | {s['usd_per_month']} $ | {s.get('note', '')} |" for s in report["subscriptions"]]
113 lines += [""] + [f"- {fact}" for fact in report["subscription_facts"]]
114 lines += ["", "## Réserves", ""] + [f"- {c}" for c in report["caveats"]]
115 return "\n".join(lines) + "\n"

118def run() -> dict:
119 bench_path = config.PROCESSED / "bench" / "report.json"
120 if not bench_path.exists():
121 raise SystemExit("Lancer d'abord `./veille bench` : le coût part des tokens mesurés.")
122 report = build_report(load_json(bench_path))
123 folder = config.PROCESSED / "cost"
124 folder.mkdir(exist_ok=True)
125 (folder / "report.json").write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
126 (config.ROOT / "docs").mkdir(exist_ok=True)
127 (config.ROOT / "docs" / "COUT.md").write_text(render_markdown(report), encoding="utf-8")
128 return report
