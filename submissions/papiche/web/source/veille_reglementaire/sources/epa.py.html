24def _fold(s: str) -> str:
25 return re.sub(r"[^a-z0-9]+", " ", (s or "").lower()).strip()

28def _sheet_rows(path):
29 """Lecture en flux de la feuille APPRIL (stdlib seule). Produit des dict colonne -> valeur."""
30 z = zipfile.ZipFile(path)
31 wb = ET.fromstring(z.read("xl/workbook.xml"))
32 rid = next(s.attrib["{http://schemas.openxmlformats.org/officeDocument/2006/relationships}id"]
33 for s in wb.iter(NS + "sheet") if s.attrib["name"] == SHEET)
34 rels = ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
35 target = next(r.attrib["Target"] for r in rels if r.attrib["Id"] == rid)
36 target = target.lstrip("/")
37 target = target if target.startswith("xl/") else "xl/" + target
38 strings = []
39 if "xl/sharedStrings.xml" in z.namelist():
40 with z.open("xl/sharedStrings.xml") as f:
41 for _, el in ET.iterparse(f):
42 if el.tag == NS + "si":
43 strings.append("".join(t.text or "" for t in el.iter(NS + "t")))
44 el.clear()
45 header = None
46 with z.open(target) as f:
47 for _, el in ET.iterparse(f):
48 if el.tag != NS + "row":
49 continue
50 vals = {}
51 for c in el.findall(NS + "c"):
52 col = re.match(r"[A-Z]+", c.attrib["r"]).group()
53 t = c.attrib.get("t")
54 if t == "inlineStr": # le dump EPA écrit ses textes en ligne
55 vals[col] = "".join(x.text or "" for x in c.iter(NS + "t"))
56 continue
57 v = c.find(NS + "v")
58 if v is None:
59 continue
60 vals[col] = strings[int(v.text)] if t == "s" else v.text
61 el.clear()
62 if header is None:
63 header = vals
64 continue
65 yield {header[k]: v for k, v in vals.items() if k in header}

79def has_current_label(product: dict) -> bool:
80 """Seuls les produits enregistrés ont une étiquette utile (les produits annulés sont ignorés)."""
81 return bool(product["label_pdf"]) and "registered" in (product["status"] or "").lower()

84def collect(scope: dict | None = None) -> dict:
85 out = config.RAW / "epa"
86 (out / "labels").mkdir(parents=True, exist_ok=True)
87 xlsx = out / "appril.xlsx"
88 polite_http.download(XLSX, xlsx) # 94 Mo une seule fois ; 304 (0 octet) si inchangé
89 scope = scope or json.loads((config.CONFIG / "scope.json").read_text())
90 wanted = [_fold(n) for p in scope["products"] for n in p.get("epa", [])]
91 rows = []
92 for r in _sheet_rows(xlsx):
93 name = _fold(r.get("PRODUCT_NAME", ""))
94 if any(w and w in name for w in wanted):
95 rows.append(r)
96 products = []
97 for r in rows:
98 labels = [x.strip() for x in (r.get("LABEL_NAMES") or "").split(",") if x.strip()]
99 products.append({"reg_no": r["REG_NUM"], "product": r["PRODUCT_NAME"], "company": r.get("COMPANY_NAME"),
100 "company_no": r.get("COMPANY_NUM"), "status": r.get("STATUS_DESC") or r.get("STATUS"),
101 "ais": r.get("AIS"), "label_pdf": labels[0] if labels else None})
102 for product in products:
103 if not has_current_label(product):
104 continue # pas de requête inutile pour un produit annulé
105 pdf = out / "labels" / product["label_pdf"]
106 if polite_http.download(PPLS + product["label_pdf"], pdf)["changed"] or not pdf.with_suffix(".txt").exists():
107 subprocess.run(["pdftotext", "-layout", str(pdf), str(pdf.with_suffix(".txt"))], check=True)
108 (out / "products.json").write_text(json.dumps({"collected_at": date.today().isoformat(), "products": products},
109 ensure_ascii=False, indent=1))
110 (out / "_meta.json").write_text(json.dumps({"collected_at": date.today().isoformat()}))
111 return {"source": "epa", "products": [(p["reg_no"], p["product"]) for p in products]}

114RATE = re.compile(r"\b(fl\.?\s?oz|oz\.?|lbs?\.?|pints?|pt\.?|quarts?|qt\.?|gal(?:lons?)?)\b[^.\n]{0,40}\b(acre|A|1,?000 sq)", re.I)

117def passages(scope: dict) -> list[Passage]:
118 meta = json.loads((config.RAW / "epa" / "products.json").read_text())
119 out: list[Passage] = []
120 for p in meta["products"]:
121 if not has_current_label(p):
122 continue # étiquettes de produits annulés : hors périmètre
123 txt = (config.RAW / "epa" / "labels" / p["label_pdf"]).with_suffix(".txt")
124 if not txt.exists():
125 continue
126 lines = [re.sub(r"\s{2,}", " ", l).strip() for l in txt.read_text(errors="replace").splitlines()]
127 lines = [l for l in lines if l]
128 seen, n_kept = set(), 0
129 for i, l in enumerate(lines):
130 if not RATE.search(l) or l in seen or n_kept >= MAX_PER_LABEL:
131 continue
132 ctx = " ".join(lines[max(0, i - 2): i + 2])[:700] # la ligne + son contexte immédiat
133 if not re.search(r"\bacre\b|1,?000 sq", ctx, re.I): # on ne garde que les taux d'application
134 continue
135 seen.add(l)
136 n_kept += 1
137 status = (p["status"] or "").lower()
138 out.append(Passage(
139 passage_id=f"epa:{p['reg_no']}:{hashlib.sha1(ctx.encode()).hexdigest()[:10]}",
140 source_id="epa", country="US", region="Amérique du Nord", product=re.sub(r"[®™]", "", p["product"]).strip().upper(),
141 reg_number=p["reg_no"], holder=p["company"] or "", status_original=p["status"] or "",
142 status="autorise" if "registered" in status else "autre",
143 ai=p.get("ais") or "", text=f"EPA Reg. No. {p['reg_no']} | Product: {p['product']} | Label text: {ctx}",
144 source_url=PPLS + p["label_pdf"], dataset_url=XLSX, collected_at=meta["collected_at"], kind="label"))
145 return out
