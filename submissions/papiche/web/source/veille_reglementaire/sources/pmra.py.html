19def collect(scope: dict | None = None) -> dict:
20 """Télécharge le registre PMRA (un seul fichier CSV) : 1 requête, et 0 octet si rien n'a changé."""
21 folder = config.RAW / "pmra"
22 result = polite_http.download(f"{API}/product", folder / "product.csv")
23 (folder / "_meta.json").write_text(json.dumps({"url": f"{API}/product", "collected_at": date.today().isoformat()}))
24 return {"source": "pmra", "url": f"{API}/product", "changed": result["changed"]}

27def _split(v: str) -> list[str]:
28 return [x.strip() for x in (v or "").split(";") if x.strip()]
29
30
31def passages(scope: dict) -> list[Passage]:
32 wanted = {n.casefold() for p in scope["products"] for n in p.get("pmra", [])}
33 meta = json.loads((config.RAW / "pmra" / "_meta.json").read_text())
34 out: list[Passage] = []
35 with open(config.RAW / "pmra" / "product.csv", encoding="cp1252", errors="replace", newline="") as f:
36 for r in csv.DictReader(f):
37 name = r["Product name - English"]
38 if name.casefold() not in wanted:
39 continue
40 reg = r["Registration number"]
41 status = r["Registration Status"]
42 pests = _split(r["Pests"]) or [""]
43 for pest in pests:
44 parts = [f"Registration number: {reg}", f"Product: {name}", f"Registration status: {status}",
45 f"Expiry date: {r['Expiry date']}", f"Registrant: {r['Registrant name']}",
46 f"Product type: {r['Product Type']}", f"Use site category: {r['Use Site Category']}"]
47 if r["Sites of Use"]:
48 parts.append(f"Sites of use: {r['Sites of Use'][:400]}")
49 parts.append(f"Pest: {pest}" if pest else "Pest: (none listed)")
50 out.append(Passage(
51 passage_id=f"pmra:{reg}:{hashlib.sha1(pest.encode()).hexdigest()[:8]}",
52 source_id="pmra", country="CA", region="Amérique du Nord", product=name, reg_number=reg,
53 holder=r["Registrant name"], status_original=status,
54 status="autorise" if status.startswith(("Full", "Conditional")) else "autre",
55 ai=r["Active ingredients - English"], text=" | ".join(parts), source_url=f"{API}/product/{reg}", dataset_url=meta["url"],
56 collected_at=meta["collected_at"],
57 gold={"dose_value": None, "dose_unit": None, "max_applications": None,
58 "pre_harvest_interval_days": None, "min_interval_days": None}))
59 return out
