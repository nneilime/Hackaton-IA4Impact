26# ── Contrat d'une source ─────────────────────────────────────────────────────
27# Une source = un module de ce dossier (nom sans « _ » initial) qui définit :
28# INFO dict : id, name, country_code, country, region, format, lang
29# collect(scope=None) télécharge les données brutes (via polite_http), retourne un petit compte rendu
30# passages(scope) retourne la liste des Passage (texte source tel quel, une unité de preuve chacun)
31# Le module est découvert automatiquement : aucune autre ligne à modifier dans le code.
32# Modèle prêt à copier : _template.py ; guide complet : docs/AJOUTER_UNE_SOURCE.md.
33
34def discover_sources() -> dict:
35 """Retourne {id: module} pour toutes les sources valides du dossier."""
36 import importlib
37 import pkgutil
38 found = {}
39 for info in pkgutil.iter_modules(__path__):
40 if info.name.startswith("_"):
41 continue
42 module = importlib.import_module(f"{__name__}.{info.name}")
43 if all(hasattr(module, attribute) for attribute in ("INFO", "collect", "passages")):
44 found[module.INFO["id"]] = module
45 return dict(sorted(found.items()))
