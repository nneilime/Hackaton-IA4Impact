41def last_collection_by_source() -> dict[str, str]:
42 dates = {}
43 for source_id in pipeline.SOURCES:
44 meta = config.RAW / source_id / "_meta.json"
45 if meta.exists():
46 dates[source_id] = json.loads(meta.read_text()).get("collected_at", "")
47 return dates

50@router.get("/overview", operation_id="adminOverview")
51def overview() -> dict:
52 records = pipeline.load("records.jsonl")
53 passages = Counter(p.source_id for p in qdrant_store.all_passages())
54 records_by_source = Counter(r["citation"]["source_id"] for r in records)
55 last_dates = last_collection_by_source()
56 queue = review.queue_for_page()
57 return {
58 "records": len(records), "by_status": dict(Counter(r["validation_status"] for r in records)),
59 "review": {"to_correct": queue["to_correct"], "to_confirm": queue["to_confirm"], "decided": queue["decided"]},
60 "sources": [{**module.INFO, "passages": passages.get(source_id, 0), "records": records_by_source.get(source_id, 0),
61 "collected_at": last_dates.get(source_id)} for source_id, module in pipeline.SOURCES.items()],
62 "system": {"model": config.MODEL, "embedding": config.EMBED_MODEL, "qdrant": qdrant_store.mode(),
63 "token_required": bool(config.ADMIN_TOKEN)},
64 }

68def history() -> list[dict]:
69 """Toutes les décisions de l'expert, de la plus récente à la plus ancienne (piste d'audit)."""
70 if not review.CORRECTIONS_FILE.exists():

94def validate_scope(scope: dict) -> dict:
95 products = scope.get("products")
96 if not isinstance(products, list) or not products:
97 raise ValueError("le périmètre doit contenir au moins un produit")
98 for product in products:
99 if not str(product.get("name", "")).strip():
100 raise ValueError("chaque produit a un nom")
101 if product.get("role") not in ROLES:
102 raise ValueError(f"rôle de « {product.get('name')} » : {', '.join(ROLES)}")
103 for key, value in product.items():
104 if key in ("role", "name"):
105 continue
106 if key not in pipeline.SOURCES or not isinstance(value, list) or not all(isinstance(n, str) for n in value):
107 raise ValueError(f"« {product['name']} » : la clé « {key} » doit être une source connue avec une liste de noms")
108 return scope

111@protected.put("/scope", operation_id="adminPutScope")
112def put_scope(body: dict) -> dict:
113 try:
114 scope = validate_scope(body)
115 except ValueError as error:
116 raise HTTPException(status_code=400, detail=str(error))
117 current = pipeline.scope()
118 scope = {"_note": current.get("_note", ""), "products": scope["products"]}
119 (config.CONFIG / "scope.json").write_text(json.dumps(scope, ensure_ascii=False, indent=2), encoding="utf-8")
120 return {"saved": True, "products": len(scope["products"])}

136class RunRequest(BaseModel):
137 task: str
138 source: Optional[str] = None
139 product: Optional[str] = None
140 per_product: Optional[int] = None
141 limit: Optional[int] = None
142
143
144def build_command(request: RunRequest) -> list[str]:
145 if request.task not in RUNNABLE_TASKS:
146 raise ValueError(f"traitement inconnu : {request.task}")
147 if request.source and request.source not in pipeline.SOURCES:
148 raise ValueError(f"source inconnue : {request.source}")
149 command = [sys.executable, "-m", "veille_reglementaire", request.task]
150 if request.source and request.task in ("collect", "index"):
151 command += ["--only", request.source]
152 if request.task == "extract":
153 if request.source:
154 command += ["--source", request.source]
155 if request.product:
156 command += ["--product", request.product]
157 if request.per_product:
158 command += ["--per-product", str(int(request.per_product))]
159 if request.limit:
160 command += ["--limit", str(int(request.limit))]
161 return command

164def read_job(job_id: str) -> dict:
165 path = JOBS_FOLDER / f"{job_id}.json"
166 if not path.exists():
167 raise HTTPException(status_code=404, detail="traitement inconnu")
168 job = json.loads(path.read_text())
169 process = RUNNING.get(job_id)
170 if job["state"] == "running" and process is not None and process.poll() is not None:
171 job.update(state="done" if process.returncode == 0 else "failed", exit_code=process.returncode,
172 finished=datetime.now(timezone.utc).isoformat(timespec="seconds"))
173 path.write_text(json.dumps(job, ensure_ascii=False))
174 RUNNING.pop(job_id, None)
175 log = (JOBS_FOLDER / f"{job_id}.log")
176 return {**job, "log": log.read_text(errors="replace")[-8000:] if log.exists() else ""}
177
178
179RUNNING: dict[str, subprocess.Popen] = {}
180
181
182@protected.post("/run", operation_id="adminRun")
183def run_task(request: RunRequest) -> dict:
184 if any(p.poll() is None for p in RUNNING.values()):
185 raise HTTPException(status_code=409, detail="un traitement est déjà en cours : attendre sa fin")
186 try:
187 command = build_command(request)
188 except ValueError as error:
189 raise HTTPException(status_code=400, detail=str(error))
190 JOBS_FOLDER.mkdir(parents=True, exist_ok=True)
191 job_id = uuid.uuid4().hex[:10]
192 log = (JOBS_FOLDER / f"{job_id}.log").open("w")
193 RUNNING[job_id] = subprocess.Popen(command, stdout=log, stderr=subprocess.STDOUT, cwd=config.ROOT)
194 job = {"id": job_id, "task": request.task, "command": " ".join(command[3:]), "state": "running",
195 "started": datetime.now(timezone.utc).isoformat(timespec="seconds")}
196 (JOBS_FOLDER / f"{job_id}.json").write_text(json.dumps(job, ensure_ascii=False))
197 return job

200@router.get("/jobs", operation_id="adminJobs")
201def list_jobs() -> list[dict]:
202 if not JOBS_FOLDER.exists():
203 return []
204 ids = [p.stem for p in sorted(JOBS_FOLDER.glob("*.json"), key=lambda p: p.stat().st_mtime, reverse=True)[:8]]
205 return [{k: v for k, v in read_job(i).items() if k != "log"} for i in ids]

208@router.get("/jobs/{job_id}", operation_id="adminJob")
209def get_job(job_id: str) -> dict:
210 return read_job(job_id)
