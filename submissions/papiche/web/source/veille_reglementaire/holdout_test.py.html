2"""Test hors échantillon : le prompt a-t-il appris la tâche, ou seulement nos 197 passages de réglage ?
3
4Même discipline qu'Arbor (jeu « dev » pour explorer, jeu « held-out » jamais vu pour valider) : on tire au hasard des
5produits e-Phy qui ne sont PAS dans config/scope.json, donc jamais regardés pendant le réglage du prompt, et on compare
6l'extraction aux colonnes structurées d'e-Phy (vérité terrain), exactement comme pour l'évaluation principale.
7"""
8from __future__ import annotations
9
10import json
11import random
12
13from . import config, evaluation, extraction_loop, pipeline
14from .sources import ephy
15
16
17def eligible_product_names() -> list[str]:
18 """Produits e-Phy qui ont des usages et qui ne sont pas dans le périmètre de réglage."""
19 tuned = {name.casefold() for product in pipeline.scope()["products"] for name in product.get("ephy", [])}
20 with_usages = {row["nom produit"] for row in ephy._rows("produits_usages_utf8.csv")}
21 return sorted(name for name in with_usages if name.casefold() not in tuned)
22
23
24def draw_passages(product_count: int, per_product: int, seed: int) -> list:
25 names = random.Random(seed).sample(eligible_product_names(), product_count)
26 passages = ephy.passages({"products": [{"ephy": [name]} for name in names]})
27 by_product: dict[str, list] = {}
28 for passage in sorted(passages, key=lambda p: p.passage_id):
29 by_product.setdefault(passage.product, []).append(passage)
30 return [p for group in by_product.values() for p in group[:per_product]]
31
32
33def count_dose_errors(traces: list[dict], expected: dict[str, dict]) -> dict:
34 """Parmi les passages publiés : dose oubliée (null alors que la source en a une) ou dose différente."""
35 omitted = wrong = 0
36 for trace in traces:
37 gold = expected.get(trace["passage"]["passage_id"])
38 if trace["outcome"] != "accepted" or not gold or gold["dose_value"] is None:
39 continue
40 extracted = trace["record"]["dose_value"]
41 omitted += extracted is None
42 wrong += extracted is not None and not evaluation.same_value(extracted, gold["dose_value"])
43 return {"dose_oubliee": omitted, "dose_differente": wrong}
44
45
46def run(product_count: int = 40, per_product: int = 4, seed: int = 1) -> dict:
47 passages = draw_passages(product_count, per_product, seed)
48 print(f"[holdout] {len(passages)} passages de {len({p.product for p in passages})} produits jamais utilisés pour le réglage", flush=True)
49 traces = []
50 for number, passage in enumerate(passages, 1):
51 traces.append(extraction_loop.extract_passage(passage))
52 print(f" {number}/{len(passages)} {passage.product[:28]:28} {traces[-1]['outcome']} essais={len(traces[-1]['attempts'])}", flush=True)
53
54 expected = {p.passage_id: p.gold for p in passages}
55 published = [t for t in traces if t["outcome"] == "accepted"]
56 report = {
57 "model": config.MODEL, "seed": seed, "products": sorted({p.product for p in passages}), "passages": len(passages),
58 "published": len(published), "sent_to_expert": len(traces) - len(published),
59 "first_try": sum(len(t["attempts"]) == 1 for t in published),
60 **evaluation.score_against_source(traces, expected), **count_dose_errors(traces, expected),
61 "passages_with_dose": sum(p.gold["dose_value"] is not None for p in passages),
62 }
63 (config.PROCESSED / "holdout.json").write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
64 (config.PROCESSED / "holdout_traces.jsonl").write_text("\n".join(json.dumps(t, ensure_ascii=False) for t in traces) + "\n", encoding="utf-8")
65 return report
