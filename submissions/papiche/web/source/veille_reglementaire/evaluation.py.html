2"""Évaluation : compare l'extraction de l'IA aux colonnes STRUCTURÉES de la source (vérité terrain e-Phy).
3
4Ce que le jury peut vérifier : exactitude champ par champ, taux de rejet, nombre d'essais, valeurs inventées.
5- PMRA ne publie pas de dose : on mesure surtout que l'IA répond bien « null ».
6- EPA (étiquettes PDF) n'a pas de vérité structurée : contrôle manuel par échantillon (voir README).
7"""
8from __future__ import annotations
9
10from collections import Counter
11
12from . import qdrant_store
13
14FIELDS_COMPARED = ["dose_value", "dose_unit", "max_applications", "pre_harvest_interval_days", "min_interval_days"]
15
16
17def same_value(extracted, expected) -> bool:
18 """Deux valeurs sont identiques si elles sont toutes deux vides, ou égales (nombres : tolérance 1e-9)."""
19 if extracted is None or expected is None:
20 return extracted is None and expected is None
21 if isinstance(extracted, float) or isinstance(expected, float):
22 return abs(extracted - expected) < 1e-9
23 return extracted == expected
24
25
26def score_against_source(traces: list[dict], expected_by_passage: dict[str, dict]) -> dict:
27 """Pour chaque passage accepté qui a une vérité terrain, compare les champs extraits aux champs attendus."""
28 correct = {field: 0 for field in FIELDS_COMPARED}
29 total = {field: 0 for field in FIELDS_COMPARED}
30 invented_values = 0 # valeur publiée alors que la source n'en contient pas
31 compared_records = 0
32
33 for trace in traces:
34 passage_id = trace["passage"]["passage_id"]
35 if trace["outcome"] != "accepted" or passage_id not in expected_by_passage:
36 continue
37 compared_records += 1
38 record, expected = trace["record"], expected_by_passage[passage_id]
39 for field in FIELDS_COMPARED:
40 total[field] += 1
41 correct[field] += same_value(record.get(field), expected.get(field))
42 if record.get(field) is not None and expected.get(field) is None:
43 invented_values += 1
44
45 return {
46 "records_evalues": compared_records,
47 "exactitude_par_champ": {f: round(correct[f] / total[f], 3) if total[f] else None for f in FIELDS_COMPARED},
48 "valeurs_inventees_publiees": invented_values,
49 }
50
51
52def run(traces: list[dict]) -> dict:
53 expected = {p.passage_id: p.gold for p in qdrant_store.all_passages() if p.gold}
54 result = score_against_source(traces, expected)
55 attempts_per_passage = Counter(len(t["attempts"]) for t in traces)
56 outcome_per_source = Counter((t["passage"]["source_id"], t["outcome"]) for t in traces)
57 result["essais_par_passage"] = dict(sorted(attempts_per_passage.items()))
58 result["issues"] = {f"{source}:{outcome}": n for (source, outcome), n in sorted(outcome_per_source.items())}
59 return result
