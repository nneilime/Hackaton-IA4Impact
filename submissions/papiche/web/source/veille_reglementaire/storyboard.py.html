2"""Storyboard vidéo de la présentation, au format de `generate_scene.sh` (Astroport.ONE/IA/generators).
3
4Une prise par diapositive Pecha Kucha : la page `index.html` est filmée vivante (animations p5.js) pendant 20 s, avec la voix-off
5du texte de `config/narration_fr.json` (voix Orpheus « amelie »). Même texte que la présentation : image et voix ne peuvent pas diverger.
6Rendu : generate_scene.sh video/storyboard.json (voir `./veille storyboard`).
7"""
8from __future__ import annotations
9
10import json
11
12from . import config, narration
13
14SLIDE_SECONDS = 20
15TRANSITION_SECONDS = 0.4
16VIDEO_FOLDER = config.ROOT / "video"
17
18
19def build(height: int = 720, voice: str = "amelie") -> dict:
20 texts = narration.render_script()
21 page = (config.ROOT / "index.html").resolve().as_uri()
22 shots = []
23 for index, text in enumerate(texts):
24 shot = {"screen": page,
25 "record": {"wait": 2500, "actions": [{"js": f"window.PechaKucha.present({index})"}, {"mark": True}]},
26 "duration": SLIDE_SECONDS, "voiceover": text}
27 if index < len(texts) - 1:
28 shot["transition"] = {"type": "fade", "duration": TRANSITION_SECONDS}
29 shots.append(shot)
30 return {"ratio": "16:9", "height": height, "style": "clean", "voice": voice, "shots": shots}
31
32
33def srt_time(seconds: float) -> str:
34 milliseconds = int(round(seconds * 1000))
35 return f"{milliseconds // 3600000:02d}:{milliseconds // 60000 % 60:02d}:{milliseconds // 1000 % 60:02d},{milliseconds % 1000:03d}"
36
37
38def build_subtitles(texts: list[str]) -> str:
39 """Sous-titres : chaque phrase de la voix-off, calée sur sa diapositive (les fondus raccourcissent la durée totale)."""
40 blocks = []
41 for index, text in enumerate(texts):
42 start = index * (SLIDE_SECONDS - TRANSITION_SECONDS) + 0.6
43 end = start + min(SLIDE_SECONDS - 2, max(4.0, len(text.split()) / 2.6)) # ≈ 2,6 mots par seconde
44 blocks.append(f"{index + 1}\n{srt_time(start)} --> {srt_time(end)}\n{text}\n")
45 return "\n".join(blocks)
46
47
48def write(height: int = 720) -> dict:
49 VIDEO_FOLDER.mkdir(exist_ok=True)
50 storyboard = build(height)
51 (VIDEO_FOLDER / "storyboard.json").write_text(json.dumps(storyboard, ensure_ascii=False, indent=2), encoding="utf-8")
52 (VIDEO_FOLDER / "sous-titres.srt").write_text(build_subtitles(narration.render_script()), encoding="utf-8")
53 return {"storyboard": "video/storyboard.json", "subtitles": "video/sous-titres.srt", "shots": len(storyboard["shots"])}
