2"""Relecture humaine : les sources que l'IA ne tranche pas, et ce que l'expert peut en faire.
3
4Deux cas arrivent dans la file :
5 - « a_corriger » : trois essais ratés, rien n'a été publié ;
6 - « a_confirmer » : publié, mais le second avis de l'IA doute.
7L'expert peut : corriger (nouvelles valeurs, contrôlées par le MÊME contrat et le MÊME contrôle de preuve que l'IA),
8confirmer une proposition, ou écarter la source (avec une raison). Chaque décision est gardée dans
9data/processed/corrections.jsonl et rejouée après chaque nouvelle extraction : une relecture n'est jamais perdue.
10Seul l'expert peut poser le statut « valide_humain ».
11"""
12from __future__ import annotations

14import json

17from pydantic import ValidationError

22CORRECTIONS_FILE = config.PROCESSED / "corrections.jsonl"
23EXTRACTION_FIELDS = list(Extraction.model_fields)

26# ── Décisions déjà prises ────────────────────────────────────────────────────

28def load_decisions() -> dict[str, dict]:
29 """Dernière décision par passage (la plus récente l'emporte)."""
30 if not CORRECTIONS_FILE.exists():
31 return {}
32 decisions = {}
33 for line in CORRECTIONS_FILE.read_text(encoding="utf-8").splitlines():
34 if line.strip():
35 entry = json.loads(line)
36 decisions[entry["passage_id"]] = entry
37 return decisions
38
39
40def save_decision(entry: dict) -> None:
41 config.PROCESSED.mkdir(parents=True, exist_ok=True)
42 with CORRECTIONS_FILE.open("a", encoding="utf-8") as file:
43 file.write(json.dumps(entry, ensure_ascii=False) + "\n")

46# ── La file à relire ─────────────────────────────────────────────────────────

48def last_proposal(trace: dict) -> dict:
49 """La dernière proposition JSON du modèle, si elle est lisible (point de départ de la correction)."""
50 for attempt in reversed(trace["attempts"]):
51 try:
52 data = json.loads(attempt["raw"] or "")
53 if isinstance(data, dict):
54 return {key: data.get(key) for key in EXTRACTION_FIELDS if key in data}
55 except ValueError:
56 continue
57 return {}
58
59
60def queue_item(trace: dict, kind: str) -> dict:
61 passage, record = trace["passage"], trace["record"]
62 if kind == "a_corriger":
63 reason = trace["attempts"][-1].get("error", "")
64 proposal = last_proposal(trace)
65 else:
66 reason = "; ".join((trace["critic"] or {}).get("motifs", [])) or "le second avis de l'IA doute"
67 proposal = {key: record.get(key) for key in EXTRACTION_FIELDS if key in record} | {"quote": record["citation"]["quote"]}
68 return {"passage_id": passage["passage_id"], "kind": kind, "product": passage["product"], "country": passage["country"],
69 "source_id": passage["source_id"], "source_url": passage["source_url"], "collected_at": passage["collected_at"],
70 "text": passage["text"], "reason": reason, "proposal": proposal, "attempts": len(trace["attempts"])}

73def review_queue(include_decided: bool = False) -> list[dict]:
74 decisions = load_decisions()
75 items = []
76 for trace in pipeline.load("traces.jsonl"):
77 if trace["outcome"] == "hitl":
78 item = queue_item(trace, "a_corriger")
79 elif trace["record"] and trace["record"]["validation_status"] == "a_confirmer":
80 item = queue_item(trace, "a_confirmer")
81 else:
82 continue
83 item["decision"] = decisions.get(item["passage_id"], {}).get("action")
84 if include_decided or item["decision"] is None:
85 items.append(item)
86 return items

89def find_trace(passage_id: str) -> dict:
90 trace = next((t for t in pipeline.load("traces.jsonl") if t["passage"]["passage_id"] == passage_id), None)
91 if trace is None:
92 raise KeyError(f"passage inconnu : {passage_id}")
93 return trace

98def build_human_record(trace: dict, fields: dict, author: str) -> dict:
99 """Valide les valeurs de l'expert comme celles de l'IA (contrat + preuve), puis fabrique l'enregistrement."""
100 passage = qdrant_store.get_passage(trace["passage"]["passage_id"])
101 if passage is None:
102 raise KeyError("texte source introuvable dans Qdrant (relancer `./veille index`)")
103 try:
104 extraction = Extraction.model_validate({k: v for k, v in fields.items() if v not in (None, "")})
105 except ValidationError as error:
106 raise ValueError(extraction_loop.format_pydantic_error(error)) from error
107 failed = [c["detail"] for c in extraction_loop.check_proof(extraction, passage.text) if not c["ok"]]
108 if failed:
109 raise ValueError("preuve non retrouvée dans la source : " + " ; ".join(failed))
110 attempts = trace["attempts"] + [{"model": f"expert:{author}"}]
111 record = extraction_loop.build_record(
112 passage, extraction, eppo_codes.normalize(extraction.crop_original, "crop"),
113 eppo_codes.normalize(extraction.target_original, "pest"), "valide_humain",
114 [f"relu et corrigé par {author}"], attempts)
115 return json.loads(record.model_dump_json())
116
117
118def write_record(record: dict) -> None:
119 """Remplace (ou ajoute) l'enregistrement dans records.jsonl et dans Qdrant."""
120 records = [r for r in pipeline.load("records.jsonl") if r["record_id"] != record["record_id"]] + [record]
121 pipeline._dump("records.jsonl", records)
122 qdrant_store.upsert_records([record])
123
124
125def apply_decision(entry: dict) -> None:
126 trace = find_trace(entry["passage_id"])
127 if entry["action"] == "corrected":
128 write_record(build_human_record(trace, entry["fields"], entry["author"]))
129 elif entry["action"] == "confirmed" and trace["record"]:
130 record = dict(trace["record"], validation_status="valide_humain",
131 critic_notes=trace["record"]["critic_notes"] + [f"confirmé par {entry['author']}"])
132 write_record(record)

135def decide(passage_id: str, action: str, author: str, fields: dict | None = None, reason: str = "") -> dict:
136 """Enregistre une décision (après l'avoir vérifiée) et l'applique tout de suite."""
137 entry = {"passage_id": passage_id, "action": action, "author": author, "reason": reason, "fields": fields or {},
138 "at": datetime.now(timezone.utc).isoformat(timespec="seconds")}
139 if action not in ("corrected", "confirmed", "dismissed"):
140 raise ValueError("action inconnue")
141 if action == "corrected" and not fields:
142 raise ValueError("aucune valeur fournie")
143 if action == "dismissed" and not reason.strip():
144 raise ValueError("une raison est obligatoire pour écarter une source")
145 apply_decision(entry) # lève une erreur claire AVANT d'enregistrer si les valeurs sont refusées
146 save_decision(entry)
147 return entry

149
150def apply_stored_corrections() -> int:
151 """À appeler après chaque extraction : rejoue les décisions de l'expert (rien n'est perdu)."""
152 count = 0
153 for entry in load_decisions().values():
154 try:
155 apply_decision(entry)
156 count += 1
157 except (KeyError, ValueError) as error:
158 print(f"[relecture] décision {entry['passage_id']} non rejouée : {error}")
159 return count

162def queue_for_page() -> dict:
163 """Résumé pour la page : file en attente et décisions déjà prises."""
164 everything = review_queue(include_decided=True)
165 waiting = [i for i in everything if i["decision"] is None]
166 shorten = lambda item: {**item, "text": item["text"][:420]}
167 return {"waiting": [shorten(i) for i in waiting], "decided": len(everything) - len(waiting),
168 "to_correct": sum(i["kind"] == "a_corriger" for i in waiting),
169 "to_confirm": sum(i["kind"] == "a_confirmer" for i in waiting)}
