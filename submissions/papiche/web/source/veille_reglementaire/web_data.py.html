17def _question_for(tr: dict) -> str:
18 r, p = tr["record"] or {}, tr["passage"]
19 bits = [p["product"]]
20 if r.get("crop_original"):
21 bits.append(f"sur {r['crop_original']}")
22 if r.get("target_original"):
23 bits.append(f"contre {r['target_original']}")
24 country = {"FR": "en France", "CA": "au Canada", "US": "aux États-Unis"}[p["country"]]
25 return f"Quelle est la dose de {' '.join(bits)} {country} ?"

28def _pick(traces: list[dict]) -> list[dict]:
29 def first(pred):
30 return next((t for t in traces if pred(t)), None)
31 cases = [
32 ("Source structurée (France, e-Phy)", first(lambda t: t["passage"]["source_id"] == "ephy" and t["outcome"] == "accepted"
33 and len(t["attempts"]) == 1 and "G46" in t["passage"]["product"])
34 or first(lambda t: t["passage"]["source_id"] == "ephy" and t["outcome"] == "accepted")),
35 ("Information absente : l'IA doit répondre null (Canada, PMRA)",
36 first(lambda t: t["passage"]["source_id"] == "pmra" and t["outcome"] == "accepted" and "M52 OD" in t["passage"]["product"])
37 or first(lambda t: t["passage"]["source_id"] == "pmra" and t["outcome"] == "accepted")),
38 ("Étiquette PDF américaine (EPA) : texte désordonné, unités US",
39 first(lambda t: t["passage"]["source_id"] == "epa" and t["outcome"] == "accepted" and (t["record"] or {}).get("dose_norm_value")
40 and t["record"]["validation_status"] == "verifie_auto")
41 or first(lambda t: t["passage"]["source_id"] == "epa" and t["outcome"] == "accepted" and (t["record"] or {}).get("dose_value"))),
42 ("La boucle corrige le modèle (retry)",
43 first(lambda t: t["outcome"] == "accepted" and len(t["attempts"]) > 1 and t["record"]["validation_status"] == "verifie_auto"
44 and t["record"].get("dose_value"))
45 or first(lambda t: t["outcome"] == "accepted" and len(t["attempts"]) > 1)),
46 ("Le modèle échoue 3 fois : arbitrage humain, rien n'est publié",
47 first(lambda t: t["outcome"] == "hitl")),
48 ]
49 out = []
50 for title, tr in cases:
51 if not tr:
52 continue
53 q = _question_for(tr) if tr["record"] else f"Quelle est la dose de {tr['passage']['product']} ?"
54 retrieval = qdrant_store.search_passages(q, k=5)
55 out.append({"title": title, "question": q, "trace": tr,
56 "retrieval": [{"score": h["score"], "passage_id": h["passage_id"], "product": h["product"],
57 "country": h["country"], "source_id": h["source_id"], "text": h["text"][:260]} for h in retrieval]})
58 return out

61def measure_performance(traces: list[dict]) -> dict:
62 """Chiffres qui expliquent la performance : ce que chaque garde-fou a rattrapé, calculé sur les vraies traces."""
63 published = [t for t in traces if t["outcome"] == "accepted"]
64 all_attempts = [a for t in traces for a in t["attempts"]]
65 return {
66 "passages": len(traces),
67 "published": len(published),
68 "published_rate": round(len(published) / len(traces), 3) if traces else 0,
69 "first_try": sum(len(t["attempts"]) == 1 for t in published),
70 "rescued_by_retry": sum(len(t["attempts"]) > 1 for t in published),
71 "sent_to_expert": len(traces) - len(published),
72 "flagged_to_confirm": sum(t["record"]["validation_status"] == "a_confirmer" for t in published),
73 "refused_by_pydantic": sum(bool(a.get("pydantic_error")) for a in all_attempts),
74 "refused_unreadable_json": sum(bool(a.get("parse_error")) for a in all_attempts),
75 "refused_by_proof_check": sum(str(a.get("error", "")).startswith("preuve non retrouvée") for a in all_attempts),
76 }

79def read_network_runs() -> dict | None:
80 """Preuve de sobriété : la collecte à froid, puis son rejeu (requêtes conditionnelles)."""
81 path = config.PROCESSED / "collect_runs.jsonl"
82 runs = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()] if path.exists() else []
83 if len(runs) >= 2:
84 return {"cold": runs[-2]["network"], "warm": runs[-1]["network"]}
85 return {"cold": runs[-1]["network"], "warm": None} if runs else None

88def count_lines(path) -> int:
89 return len(path.read_text(encoding="utf-8").splitlines()) if path.exists() else 0

92def read_prompt_history() -> list[dict]:
93 """Les trois versions du prompt d'extraction, relues dans data/processed/history (preuve de la démarche)."""
94 history = config.PROCESSED / "history"
95 versions = [("v1", "Prompt de départ", history / "prompt_v1"),
96 ("v2", "Contrôle de preuve plus strict, ancien prompt", history / "prompt_v2_strict"),
97 ("v3", "Règle de citation précisée (version actuelle)", config.PROCESSED)]
98 rows = []
99 for name, description, folder in versions:
100 if not (folder / "eval.json").exists():
101 continue
102 evaluation = json.loads((folder / "eval.json").read_text())
103 rows.append({"version": name, "description": description, "published": count_lines(folder / "records.jsonl"),
104 "sent_to_expert": count_lines(folder / "hitl.jsonl"),
105 "first_try": evaluation["essais_par_passage"].get("1", 0)})
106 return rows

109def build():
110 traces = pipeline.load("traces.jsonl")
111 records = pipeline.load("records.jsonl")
112 passages = qdrant_store.all_passages()
113 qa = []
114 for q in QUESTIONS:
115 r = question_answering.ask(q)
116 qa.append({k: r[k] for k in ("question", "plan", "method", "answer", "answer_grounded_by_code", "citations")}
117 | {"rows": r["rows"][:6]})
118 data = {
119 "built": date.today().isoformat(), "model": config.MODEL, "embed_model": config.EMBED_MODEL,
120 "llm_backend": llm.backend(), "qdrant": qdrant_store.mode(),
121 "counts": {"passages": len(passages), "passages_by_source": dict(Counter(p.source_id for p in passages)),
122 "records": len(records), "hitl": len(pipeline.load("hitl.jsonl")),
123 "retries": sum(1 for t in traces if len(t["attempts"]) > 1),
124 "status": dict(Counter(r["validation_status"] for r in records))},
125 "eval": json.loads((config.PROCESSED / "eval.json").read_text()) if (config.PROCESSED / "eval.json").exists() else None,
126 "prompts": {n: (config.PROMPTS / n).read_text(encoding="utf-8")
127 for n in ("extractor.system.md", "retry.user.md", "critic.system.md", "eppo.system.md", "query.system.md")},
128 "bench": json.loads((config.PROCESSED / "bench" / "report.json").read_text()) if (config.PROCESSED / "bench" / "report.json").exists() else None,
129 "cost": json.loads((config.PROCESSED / "cost" / "report.json").read_text()) if (config.PROCESSED / "cost" / "report.json").exists() else None,
130 "code": source_viewer.build(), "review": review.queue_for_page(),
131 "sources": [dict(module.INFO) for module in pipeline.SOURCES.values()],
132 "holdout": json.loads((config.PROCESSED / "holdout.json").read_text()) if (config.PROCESSED / "holdout.json").exists() else None,
133 "narration": narration.render_script(), "audio_available": (narration.AUDIO_FOLDER / "pk-01.mp3").exists(), "network": read_network_runs(), "performance": measure_performance(traces), "prompt_history": read_prompt_history(),
134 "models_py": (config.ROOT / "veille_reglementaire" / "models.py").read_text(encoding="utf-8"),
135 "cases": _pick(traces), "qa": qa,
136 "table": comparison_table.table(records), "footprint": comparison_table.footprint(records),
137 "sisters": pipeline.load("sisters.jsonl"),
138 "scope": pipeline.scope(),
139 }
140 (config.WEB / "data.js").write_text("window.VEILLE = " + json.dumps(data, ensure_ascii=False) + ";\n", encoding="utf-8")
141 print(f"web/data.js écrit ({len(json.dumps(data)) // 1024} Ko)")
