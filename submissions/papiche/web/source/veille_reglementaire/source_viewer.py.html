65def find_symbol_lines(source: str, symbol: str) -> tuple[int, int]:
66 """Lignes (début, fin) d'une fonction, d'une classe ou d'une méthode « Classe.methode »."""
67 tree = ast.parse(source)
68 scope = tree.body
69 node = None
70 for name in symbol.split("."):
71 node = next(n for n in scope if getattr(n, "name", None) == name)
72 scope = getattr(node, "body", [])
73 start = min([node.lineno] + [d.lineno for d in getattr(node, "decorator_list", [])])
74 return start, node.end_lineno

77def highlighted_lines(source: str, file_name: str) -> list[str]:
78 try:
79 lexer = get_lexer_for_filename(file_name)
80 except Exception:
81 lexer = TextLexer()
82 return highlight(source, lexer, FORMATTER).rstrip("\n").split("\n")

87def source_url(path: str, start: int | None = None, end: int | None = None) -> str:
88 anchor = f"#L{start}-L{end}" if start else ""
89 return f"web/source/{path}.html{anchor}"

92def build_references() -> dict:
93 references = {}
94 for key, (path, symbol, title) in REFERENCES.items():
95 source = (config.ROOT / path).read_text(encoding="utf-8")
96 total_lines = len(source.splitlines())
97 start, end = find_symbol_lines(source, symbol) if symbol else (1, total_lines)
98 lines = highlighted_lines(source, path)
99 numbered = "\n".join(f'<span class="cl"><i>{n}</i>{lines[n - 1]}</span>' for n in range(start, end + 1))
100 references[key] = {"title": title, "path": path, "symbol": symbol, "start": start, "end": end,
101 "url": source_url(path, start, end), "html": numbered}
102 return references

123def write_source_page(path: str) -> str:
124 source = (config.ROOT / path).read_text(encoding="utf-8")
125 lines = highlighted_lines(source, path)
126 depth = path.count("/") + 2 # web/source/<chemin>.html -> racine du dépôt
127 back = "../" * depth + "index.html"
128 rows = "\n".join(f'<tr id="L{n}"><td class="ln"><a href="#L{n}">{n}</a></td><td class="hl">{line}</td></tr>'
129 for n, line in enumerate(lines, 1))
130 page = (f'<!doctype html><html lang="fr"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">'
131 f'<title>{html.escape(path)}</title><link rel="stylesheet" href="{"../" * path.count("/")}pygments.css"><style>{PAGE_STYLE}</style>'
132 f'<header><a href="{back}">← retour à la présentation</a><span class="path">{html.escape(path)}</span>'
133 f'<a href="{"../" * path.count("/")}index.html">tout le code</a><span class="licence">AGPL-3.0-or-later</span></header>'
134 f'<table>{rows}</table><script>{HIGHLIGHT_SCRIPT}</script></html>')
135 target = SOURCE_FOLDER / f"{path}.html"
136 target.parent.mkdir(parents=True, exist_ok=True)
137 target.write_text(page, encoding="utf-8")
138 return path

141def published_paths() -> list[str]:
142 paths = []
143 for folder in PUBLISHED_FOLDERS:
144 paths += sorted(str(p.relative_to(config.ROOT)) for p in (config.ROOT / folder).rglob("*")
145 if p.is_file() and p.suffix in (".py", ".md") and "__pycache__" not in p.parts)
146 return paths + [p for p in PUBLISHED_FILES if (config.ROOT / p).exists()]

149def first_docstring_line(path: str) -> str:
150 if not path.endswith(".py"):
151 return ""
152 docstring = ast.get_docstring(ast.parse((config.ROOT / path).read_text(encoding="utf-8"))) or ""
153 return docstring.splitlines()[0] if docstring else ""

156def write_index_page(paths: list[str]) -> None:
157 items = "\n".join(f'<li><a href="{p}.html">{html.escape(p)}</a> <span>{html.escape(first_docstring_line(p))}</span></li>' for p in paths)
158 page = (f'<!doctype html><html lang="fr"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">'
159 f'<title>Tout le code</title><style>{PAGE_STYLE}</style><header><a href="../../index.html">← retour à la présentation</a>'
160 f'<span class="path">tout le code · {len(paths)} fichiers</span><span class="licence">AGPL-3.0-or-later</span></header>'
161 f'<ul class="index" style="padding:14px 34px">{items}</ul></html>')
162 (SOURCE_FOLDER / "index.html").write_text(page, encoding="utf-8")

165def build() -> dict:
166 """Écrit les pages de source et retourne les extraits pour la page d'accueil."""
167 SOURCE_FOLDER.mkdir(parents=True, exist_ok=True)
168 (SOURCE_FOLDER / "pygments.css").write_text(HtmlFormatter(style="monokai").get_style_defs(".hl"), encoding="utf-8")
169 paths = published_paths()
170 for path in paths:
171 write_source_page(path)
172 write_index_page(paths)
173 return build_references()
