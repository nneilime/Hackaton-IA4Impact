2"""Boucle de contrôle : extraire (IA) → valider la forme (Pydantic) → vérifier la preuve (code) → recommencer.
3
4Principe : l'IA propose, le code décide. Un enregistrement n'est publié que si chaque valeur est retrouvée
5mot à mot dans le passage source. Après MAX_ATTEMPTS échecs, rien n'est publié : un expert tranche.
6"""

11import re
12from datetime import date

14from pydantic import ValidationError

16from . import config, eppo_codes, llm
17from .models import Citation, CriticVerdict, Extraction, Record, normalize_dose, numbers_in
18from .sources import Passage
19
20FIELDS_COPIED_FROM_PASSAGE = ("crop_original", "target_original", "dose_original", "dose_unit")
21COUNTED_FIELDS = ("max_applications", "min_interval_days", "pre_harvest_interval_days")

24# ── Petits outils ────────────────────────────────────────────────────────────

26def read_prompt(file_name: str) -> str:
27 return (config.PROMPTS / file_name).read_text(encoding="utf-8")

30def normalize_text(text: str | None) -> str:
31 """Compare sans tenir compte des espaces multiples ni de la casse (jamais des mots)."""
32 return re.sub(r"\s+", " ", text or "").strip().casefold()
33
34
35def build_user_prompt(passage: Passage) -> str:
36 return (f"Source : {passage.source_id} ({passage.country}) — produit {passage.product}\n"
37 f"<passage>\n{passage.text}\n</passage>\n"
38 "Extrais l'enregistrement JSON.")
39
40
41def build_retry_prompt(first_prompt: str, previous_answer: str | None, error: str) -> str:
42 """Le modèle revoit sa réponse refusée et le motif exact du refus."""
43 return (f"{first_prompt}\n\nTa réponse précédente :\n{previous_answer or '(vide)'}\n\n"
44 + read_prompt("retry.user.md").format(error=error))

46
47def format_pydantic_error(error: ValidationError) -> str:
48 return "; ".join(f"{'.'.join(map(str, e['loc'])) or 'objet'}: {e['msg']}" for e in error.errors())

51# ── Contrôle de la preuve (en code : rien n'est laissé à l'appréciation de l'IA) ─
52
53def check_proof(extraction: Extraction, passage_text: str) -> list[dict]:
54 """Chaque valeur extraite doit exister dans le passage. Retourne la liste des contrôles et leur résultat."""
55 passage = normalize_text(passage_text)
56 checks: list[dict] = []
57
58 def add(name: str, passed: bool, detail: str):
59 checks.append({"check": name, "ok": passed, "detail": detail})
60
61 add("citation_retrouvee", normalize_text(extraction.quote) in passage,
62 f"« {extraction.quote} » doit figurer mot à mot dans le passage")
63
64 for field in FIELDS_COPIED_FROM_PASSAGE:
65 value = getattr(extraction, field)
66 if value:
67 add(f"{field}_retrouve", normalize_text(value) in passage, f"{field}={value!r} doit figurer dans le passage")
68
69 for restriction in extraction.restrictions:
70 add("restriction_retrouvee", normalize_text(restriction) in passage,
71 f"restriction {restriction!r} doit figurer dans le passage")
72
73 numbers_in_passage = numbers_in(passage_text)
74 for field in COUNTED_FIELDS:
75 value = getattr(extraction, field)
76 if value is not None:
77 add(f"{field}_present", float(value) in numbers_in_passage, f"{field}={value} doit apparaître dans le passage")
78
79 if extraction.dose_value is not None:
80 add("dose_portee_par_la_citation", normalize_text(extraction.dose_original) in normalize_text(extraction.quote),
81 f"la citation doit contenir la dose « {extraction.dose_original} »")
82 return checks

85def ask_critic(passage_text: str, extraction: Extraction) -> dict:
86 """Second avis de l'IA, indicatif : il peut dégrader le statut, jamais publier à la place du code."""
87 answer = llm.chat_json(read_prompt("critic.system.md"),
88 f"<passage>\n{passage_text}\n</passage>\nEnregistrement :\n{extraction.model_dump_json()}",
89 num_predict=200)
90 try:
91 verdict = CriticVerdict.model_validate(json.loads(answer["text"] or "{}"))
92 return {"verdict": verdict.verdict, "motifs": verdict.motifs, "raw": answer["text"], "cached": answer["cached"]}
93 except ValueError as e:
94 return {"verdict": "indetermine", "motifs": [f"sortie du critique invalide : {e}"],
95 "raw": answer["text"], "cached": answer["cached"]}

98# ── Une tentative d'extraction ───────────────────────────────────────────────
99
100def run_one_attempt(number: int, system_prompt: str, user_prompt: str, passage: Passage) -> tuple[dict, Extraction | None]:
101 """Appelle l'IA, valide la forme, vérifie la preuve. Retourne (compte rendu de l'essai, extraction si acceptée)."""
102 answer = llm.chat_json(system_prompt, user_prompt)
103 attempt = {"n": number, "user_prompt": user_prompt, "raw": answer["text"], "cached": answer["cached"],
104 "model": answer["model"], "backend": answer["backend"], "parse_error": None, "pydantic_error": None,
105 "checks": [], "accepted": False}
106
107 extraction, error = None, None
108 try:
109 data = json.loads(answer["text"] or "")
110 if not isinstance(data, dict):
111 raise ValueError("la racine doit être un objet JSON")
112 extraction = Extraction.model_validate(data)
113 except ValidationError as e: # la forme ne respecte pas le contrat Pydantic
114 error = attempt["pydantic_error"] = format_pydantic_error(e)
115 except ValueError as e: # JSON illisible (JSONDecodeError est un ValueError)
116 error = attempt["parse_error"] = f"JSON invalide : {e}"
117
118 if extraction is not None:
119 attempt["checks"] = check_proof(extraction, passage.text)
120 failed = [c for c in attempt["checks"] if not c["ok"]]
121 if failed:
122 error = "preuve non retrouvée : " + " ; ".join(c["detail"] for c in failed)
123
124 if error:
125 attempt["error"] = error
126 return attempt, None
127 attempt["accepted"] = True
128 return attempt, extraction

133def build_record(passage: Passage, extraction: Extraction, crop_eppo: dict, target_eppo: dict,
134 status: str, notes: list[str], attempts: list[dict]) -> Record:
135 dose_norm_value, dose_norm_unit = normalize_dose(extraction.dose_value, extraction.dose_unit)
136 citation = Citation(
137 source_id=passage.source_id, source_url=passage.source_url, dataset_url=passage.dataset_url,
138 collected_at=date.fromisoformat(passage.collected_at), quote=extraction.quote,
139 passage_id=passage.passage_id, passage_sha256=hashlib.sha256(passage.text.encode()).hexdigest())
140 return Record(
141 record_id=hashlib.sha1(passage.passage_id.encode()).hexdigest()[:16],
142 product=passage.product, country=passage.country, region=passage.region, reg_number=passage.reg_number,
143 holder=passage.holder, authorization_status_original=passage.status_original,
144 authorization_status=passage.status,
145 crop_original=extraction.crop_original, crop_eppo=crop_eppo["eppo"],
146 target_original=extraction.target_original, target_eppo=target_eppo["eppo"],
147 dose_original=extraction.dose_original, dose_value=extraction.dose_value, dose_unit=extraction.dose_unit,
148 dose_norm_value=dose_norm_value, dose_norm_unit=dose_norm_unit,
149 max_applications=extraction.max_applications, min_interval_days=extraction.min_interval_days,
150 pre_harvest_interval_days=extraction.pre_harvest_interval_days, restrictions=extraction.restrictions,
151 citation=citation, validation_status=status, attempts=len(attempts),
152 model=attempts[-1]["model"], critic_notes=notes)

155def extract_passage(passage: Passage, max_attempts: int | None = None, use_llm_critic: bool | None = None) -> dict:
156 """Traite un passage. Retourne une trace complète (utilisée par la page pédagogique) et le Record si accepté."""
157 max_attempts = max_attempts or config.MAX_ATTEMPTS
158 use_llm_critic = config.LLM_CRITIC if use_llm_critic is None else use_llm_critic
159 system_prompt = read_prompt("extractor.system.md")
160 first_prompt = build_user_prompt(passage)
161
162 attempts: list[dict] = []
163 extraction = None
164 user_prompt = first_prompt
165 for number in range(1, max_attempts + 1):
166 attempt, extraction = run_one_attempt(number, system_prompt, user_prompt, passage)
167 attempts.append(attempt)
168 if extraction is not None:
169 break
170 user_prompt = build_retry_prompt(first_prompt, attempt["raw"], attempt["error"])
171
172 trace = {
173 "passage": {key: getattr(passage, key) for key in
174 ("passage_id", "source_id", "country", "product", "reg_number", "text", "source_url", "collected_at")},
175 "attempts": attempts, "outcome": "accepted" if extraction else "hitl",
176 "record": None, "critic": None, "eppo": {},
177 }
178 if extraction is None:
179 return trace # trois échecs : rien n'est publié, un expert tranche
180
181 status, notes = "verifie_auto", []
182 if use_llm_critic:
183 trace["critic"] = ask_critic(passage.text, extraction)
184 if trace["critic"]["verdict"] != "ok":
185 status = "a_confirmer"
186 notes = ["second avis IA : " + ("; ".join(trace["critic"]["motifs"]) or trace["critic"]["verdict"])]
187
188 crop_eppo = eppo_codes.normalize(extraction.crop_original, "crop")
189 target_eppo = eppo_codes.normalize(extraction.target_original, "pest")
190 trace["eppo"] = {"crop": crop_eppo, "target": target_eppo}
191 record = build_record(passage, extraction, crop_eppo, target_eppo, status, notes, attempts)
192 trace["record"] = json.loads(record.model_dump_json())
193 return trace
