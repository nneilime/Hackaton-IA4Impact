2"""Normalisation EPPO : le LLM propose, la table confirme. Code inconnu -> null, jamais deviné.
3
4config/eppo_seed.csv est une AMORCE de codes vérifiés à la main ; LPC la remplace par son jeu EPPO officiel
5(même format : code,kind,names). La valeur originale reste toujours conservée à côté du code.
6"""
7from __future__ import annotations
8import csv, difflib, functools, json, re, unicodedata
9from . import config, llm
10from .models import EppoChoice

13def fold(s: str) -> str:
14 s = unicodedata.normalize("NFKD", s or "")
15 s = "".join(c for c in s if not unicodedata.combining(c)).casefold()
16 return re.sub(r"\s+", " ", s).strip()
17
18
19@functools.lru_cache(maxsize=1)
20def table() -> dict[str, list[tuple[str, str]]]:
21 """kind -> [(nom_normalisé, code)]"""
22 t: dict[str, list[tuple[str, str]]] = {"crop": [], "pest": []}
23 with open(config.CONFIG / "eppo_seed.csv", encoding="utf-8") as f:
24 for r in csv.DictReader(f):
25 for n in r["names"].split("|"):
26 t[r["kind"]].append((fold(n), r["code"]))
27 return t

30def _parts(original: str) -> list[str]:
31 return [p.strip() for p in re.split(r"\s+-\s+|,|;|/|\s+et\s+|\s+and\s+", original) if p.strip()]
32
33
34def candidates(name: str, kind: str, n: int = 5) -> list[tuple[str, str, float]]:
35 f = fold(name)
36 scored = {}
37 for nm, code in table()[kind]:
38 r = 1.0 if nm == f else difflib.SequenceMatcher(None, f, nm).ratio()
39 if nm in f or f in nm:
40 r = max(r, 0.8)
41 if code not in scored or r > scored[code][1]:
42 scored[code] = (nm, r)
43 top = sorted(((c, nm, r) for c, (nm, r) in scored.items()), key=lambda x: -x[2])[:n]
44 return [(c, nm, r) for c, nm, r in top if r >= 0.6]

47def normalize(original: str | None, kind: str) -> dict:
48 """Retourne {"eppo": "A;B"|None, "method": ..., "steps": [...]} (steps = trace pédagogique)."""
49 if not original:
50 return {"eppo": None, "method": "absent", "steps": []}
51 codes, steps = [], []
52 for part in _parts(original):
53 cands = candidates(part, kind)
54 exact = [c for c, nm, r in cands if r == 1.0]
55 if exact:
56 codes.append(exact[0]); steps.append({"part": part, "method": "table", "code": exact[0], "candidates": [c for c, *_ in cands]})
57 continue
58 if not cands:
59 steps.append({"part": part, "method": "aucun_candidat", "code": None, "candidates": []})
60 return {"eppo": None, "method": "non_confirme", "steps": steps}
61 listing = "\n".join(f"- {c} ({nm})" for c, nm, _ in cands)
62 res = llm.chat_json(config.PROMPTS.joinpath("eppo.system.md").read_text(),
63 f"Nom à normaliser : « {part} »\nCandidats :\n{listing}", num_predict=60)
64 code = None
65 try:
66 code = EppoChoice.model_validate(json.loads(res["text"] or "{}")).eppo
67 except Exception:
68 pass
69 ok = code in {c for c, *_ in cands} # le LLM ne peut choisir que dans la liste
70 steps.append({"part": part, "method": "llm_parmi_candidats", "code": code if ok else None,
71 "candidates": [c for c, *_ in cands], "llm_raw": res["text"]})
72 if not ok or code is None:
73 return {"eppo": None, "method": "non_confirme", "steps": steps}
74 codes.append(code)
75 return {"eppo": ";".join(dict.fromkeys(codes)) or None, "method": "confirme", "steps": steps}
