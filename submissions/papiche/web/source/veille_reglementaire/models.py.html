11_NUM = re.compile(r"\d+(?:[.,]\d+)?")
12
13
14def numbers_in(text: str) -> set[float]:
15 """Tous les nombres d'un texte ('1,25' et '1.25' valent 1.25)."""
16 return {float(n.replace(",", ".")) for n in _NUM.findall(text or "")}

19class Extraction(BaseModel):
20 """Ce que le modèle de langage doit remplir, et RIEN d'autre (champs absents = null)."""
21 model_config = ConfigDict(extra="forbid")
22
23 crop_original: Optional[str] = None # culture telle qu'écrite dans la source
24 target_original: Optional[str] = None # cible (ravageur / maladie) telle qu'écrite
25 dose_original: Optional[str] = None # dose telle qu'écrite : "0.5 kg/ha"
26 dose_value: Optional[float] = Field(default=None, gt=0)
27 dose_unit: Optional[str] = None # unité telle qu'écrite : "kg/ha", "g/m3"...
28 max_applications: Optional[int] = Field(default=None, ge=1)
29 min_interval_days: Optional[int] = Field(default=None, ge=0)
30 pre_harvest_interval_days: Optional[int] = Field(default=None, ge=0)
31 restrictions: list[str] = Field(default_factory=list) # extraits littéraux
32 quote: str = Field(min_length=6) # preuve : phrase copiée mot à mot
33
34 @model_validator(mode="after")
35 def dose_coherente(self):
36 if self.dose_value is not None:
37 if not self.dose_original or not self.dose_unit:
38 raise ValueError("dose_value exige dose_original et dose_unit")
39 if self.dose_value not in numbers_in(self.dose_original):
40 raise ValueError(f"dose_value={self.dose_value} absente de dose_original={self.dose_original!r}")
41 if len(self.dose_unit) > 30 or re.search(r"[\[\]()]", self.dose_unit):
42 raise ValueError(f"dose_unit={self.dose_unit!r} n'est pas une unité (court, sans crochets ni parenthèses)")
43 if self.dose_unit.replace(" ", "").casefold() not in self.dose_original.replace(" ", "").casefold():
44 raise ValueError(f"dose_unit={self.dose_unit!r} absente de dose_original={self.dose_original!r}")
45 elif self.dose_original or self.dose_unit:
46 raise ValueError("dose_original/dose_unit renseignés sans dose_value")
47 return self

123_UNITS = { # unité (sans espaces ni points, minuscule) -> (facteur, unité normalisée)
124 "kg/ha": (1, "kg/ha"), "g/ha": (1e-3, "kg/ha"), "l/ha": (1, "L/ha"), "ml/ha": (1e-3, "L/ha"),
125 # unités américaines (EPA) : 1 acre = 0,404686 ha
126 "lb/acre": (0.453592 / 0.404686, "kg/ha"), "lbs/acre": (0.453592 / 0.404686, "kg/ha"),
127 "oz/acre": (0.0283495 / 0.404686, "kg/ha"),
128 "floz/acre": (0.0295735 / 0.404686, "L/ha"), "pt/acre": (0.473176 / 0.404686, "L/ha"),
129 "pint/acre": (0.473176 / 0.404686, "L/ha"), "qt/acre": (0.946353 / 0.404686, "L/ha"),
130 "quart/acre": (0.946353 / 0.404686, "L/ha"), "gal/acre": (3.78541 / 0.404686, "L/ha"),
131}

134def normalize_dose(value: Optional[float], unit: Optional[str]):
135 """Conversion déterministe vers kg/ha ou L/ha ; toute autre unité reste non normalisée (null).
136 La valeur originale est toujours conservée à côté (dose_original, dose_value, dose_unit)."""
137 if value is None or not unit:
138 return None, None
139 u = re.sub(r"[\s.]", "", unit).casefold()
140 if u in _UNITS:
141 f, nu = _UNITS[u]
142 return round(value * f, 4), nu
143 return None, None
