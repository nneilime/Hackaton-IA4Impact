8def main(argv=None):
9 ap = argparse.ArgumentParser(prog="veille_reglementaire", description="Agent IA maison de veille réglementaire LPC")
10 sub = ap.add_subparsers(dest="cmd", required=True)
11 sub.add_parser("status", help="modèle, backend, Qdrant, volumes")
12 c = sub.add_parser("collect", help="télécharge les sources officielles"); c.add_argument("--only", nargs="*")

14 e = sub.add_parser("extract", help="boucle IA : extraire, valider, auditer")
15 e.add_argument("--limit", type=int); e.add_argument("--product"); e.add_argument("--source")
16 e.add_argument("--per-product", type=int, help="échantillon : N passages max par produit et pays")
17 sub.add_parser("eval", help="évalue contre la vérité structurée")
18 t = sub.add_parser("compare", help="tableau produit x pays x usage"); t.add_argument("--product"); t.add_argument("--csv")
19 sub.add_parser("sisters", help="hypothèses de produits sœurs")
20 a = sub.add_parser("ask", help="question en langage naturel"); a.add_argument("question")
21 s = sub.add_parser("serve", help="API REST + interface d'administration (FastAPI)"); s.add_argument("--port", type=int, default=8088)
22 s.add_argument("--host", default="127.0.0.1", help="127.0.0.1 par défaut (local seulement) ; 0.0.0.0 pour ouvrir au réseau")
23 sub.add_parser("web", help="génère web/data.js pour la page didactique")
24 b = sub.add_parser("bench", help="comparatif de modèles LLM + dimensionnement GPU / stockage")
25 b.add_argument("--models", nargs="*"); b.add_argument("--small", action="store_true", help="échantillon réduit (12 passages)")
26 sub.add_parser("cost", help="coût : service local vs API au token vs forfaits (après `bench`)")
27 sub.add_parser("network", help="requêtes sortantes vers les sources (preuve de sobriété)")
28 v = sub.add_parser("voice", help="génère les fichiers audio de la présentation (web/audio/)")
29 v.add_argument("--engine", default="orpheus", choices=["orpheus", "espeak"]); v.add_argument("--voice", default="amelie")
30 v.add_argument("--slides", type=int, nargs="*", help="refaire seulement ces diapositives (ex. --slides 13)")
31 h = sub.add_parser("holdout", help="test hors échantillon : produits e-Phy jamais utilisés pour régler le prompt")
32 h.add_argument("--products", type=int, default=40); h.add_argument("--per-product", type=int, default=4); h.add_argument("--seed", type=int, default=1)
33 n = sub.add_parser("new-source", help="crée le squelette d'une nouvelle source officielle")
34 n.add_argument("source_id"); n.add_argument("--name", required=True); n.add_argument("--country-code", required=True)
35 n.add_argument("--country", required=True); n.add_argument("--region", required=True)
36 n.add_argument("--format", default="CSV"); n.add_argument("--lang", default="EN")
37 r = sub.add_parser("review", help="sources à faire relire par un humain (trois échecs, ou « à confirmer »)")
38 r.add_argument("--all", action="store_true", help="inclure les sources déjà décidées"); r.add_argument("--show", metavar="N_OU_ID")
39 for name, text in (("confirm", "confirmer la proposition de l'IA"), ("correct", "corriger avec de nouvelles valeurs"), ("dismiss", "écarter la source")):
40 d = sub.add_parser(name, help=text); d.add_argument("target", metavar="N_OU_ID"); d.add_argument("--author", required=True)
41 if name == "dismiss":
42 d.add_argument("--reason", required=True)
43 if name == "correct":
44 for flag in ("crop-original", "target-original", "dose-original", "dose-unit", "quote"):
45 d.add_argument(f"--{flag}")
46 for flag in ("dose-value",):
47 d.add_argument(f"--{flag}", type=float)
48 for flag in ("max-applications", "min-interval-days", "pre-harvest-interval-days"):
49 d.add_argument(f"--{flag}", type=int)
50 sub.add_parser("storyboard", help="écrit video/storyboard.json (format generate_scene.sh) et les sous-titres")
51 sub.add_parser("export", help="CSV comparatif + contrat OpenAPI + schéma Dataverse (dossier m365/)")
52 args = ap.parse_args(argv)
53
54 if args.cmd == "status":
55 print(json.dumps({"modele": config.MODEL, "embed": config.EMBED_MODEL, "backend_llm": llm.backend(),
56 "qdrant": qdrant_store.mode(), "records": len(pipeline.load("records.jsonl")),
57 "hitl": len(pipeline.load("hitl.jsonl"))}, ensure_ascii=False, indent=1))
58 elif args.cmd == "collect":
59 print(json.dumps(pipeline.collect(args.only), ensure_ascii=False, indent=1))
60 elif args.cmd == "index":
61 pipeline.index(args.only)
62 elif args.cmd == "extract":
63 print(json.dumps(pipeline.extract(args.limit, args.product, args.source, args.per_product)))
64 elif args.cmd == "eval":
65 from . import evaluation
66 r = evaluation.run(pipeline.load("traces.jsonl"))
67 (config.PROCESSED / "eval.json").write_text(json.dumps(r, ensure_ascii=False, indent=1))
68 print(json.dumps(r, ensure_ascii=False, indent=1))
69 elif args.cmd == "compare":
70 rows = comparison_table.table(pipeline.load("records.jsonl"), args.product)
71 if args.csv:
72 open(args.csv, "w", encoding="utf-8").write(comparison_table.to_csv(rows))
73 for r in rows[:40]:
74 print(f"{r['product'][:20]:20} {r['country']} {str(r['crop_original'])[:22]:22} {str(r['target_original'])[:24]:24} "
75 f"{str(r['dose_original'])[:22]:22} {r['validation_status']}")
76 print(f"{len(rows)} lignes")
77 elif args.cmd == "sisters":
78 from . import sister_products
79 meta = [{"product": p.product, "country": p.country, "ai": p.ai, "holder": p.holder} for p in qdrant_store.all_passages()]
80 found = sister_products.detect(meta)
81 pipeline._dump("sisters.jsonl", found)
82 for s_ in found:
83 print(f"{s_['confidence']:.2f} {s_['product_a']} ({s_['country_a']}) ~ {s_['product_b']} ({s_['country_b']})")
84 elif args.cmd == "ask":
85 from . import question_answering
86 r = question_answering.ask(args.question)
87 print(r["answer"]); [print(f" [{c['n']}] {c['source_url']} ({c['collected_at']})") for c in r["citations"][:6]]
88 elif args.cmd == "serve":
89 import uvicorn
90 uvicorn.run("veille_reglementaire.api:app", host=args.host, port=args.port)
91 elif args.cmd == "bench":
92 from . import model_benchmark
93 r = model_benchmark.run(args.models, {"ephy": 6, "pmra": 3, "epa": 3} if args.small else None)
94 print(json.dumps({m: {k: v for k, v in x.items() if k in ("s_per_passage_mean", "decode_tok_s", "accepted", "hitl", "accuracy_vs_structured")}
95 for m, x in r["results"].items()}, ensure_ascii=False, indent=1))
96 elif args.cmd == "cost":
97 from . import cost_estimate
98 report = cost_estimate.run()
99 print(f"docs/COUT.md écrit ({len(report['options'])} options, {len(report['scenarios'])} scénarios)")
100 elif args.cmd == "network":
101 from . import polite_http
102 summary = polite_http.summarize_requests(polite_http.read_request_log())
103 print(json.dumps(summary, ensure_ascii=False, indent=1))
104 elif args.cmd == "voice":
105 from . import narration
106 report = narration.render_audio(args.engine, args.voice, args.slides)
107 too_long = [r["slide"] for r in report if not r["fits_20s"]]
108 print("Toutes les diapositives tiennent en 20 s." if not too_long else f"Trop longues (>20 s) : {too_long}")
109 elif args.cmd == "holdout":
110 from . import holdout_test
111 print(json.dumps(holdout_test.run(args.products, args.per_product, args.seed), ensure_ascii=False, indent=1))
112 elif args.cmd == "new-source":
113 from . import source_scaffold
114 path = source_scaffold.create_source(args.source_id, args.name, args.country_code, args.country, args.region, args.format, args.lang)
115 print(f"Créé : {path}\n" + source_scaffold.NEXT_STEPS.format(path=path, source_id=args.source_id))
116 elif args.cmd in ("review", "confirm", "correct", "dismiss"):
117 from . import review_cli
118 review_cli.run(args)
119 elif args.cmd == "storyboard":
120 from . import storyboard
121 print(json.dumps(storyboard.write(), ensure_ascii=False))
122 elif args.cmd == "export":
123 from . import microsoft365_export
124 microsoft365_export.export()
125 elif args.cmd == "web":
126 from . import web_data
127 web_data.build()
128
129
130if __name__ == "__main__":
131 main()
