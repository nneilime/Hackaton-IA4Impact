2"""Crée le squelette d'une nouvelle source à partir du modèle `sources/_template.py`."""
3from __future__ import annotations
4
5import re
6
7from . import config
8
9SOURCES_FOLDER = config.ROOT / "veille_reglementaire" / "sources"
10
11
12def create_source(source_id: str, name: str, country_code: str, country: str, region: str,
13 data_format: str = "CSV", language: str = "EN") -> str:
14 """Écrit sources/<id>.py et retourne son chemin. Refuse un identifiant invalide ou déjà pris."""
15 if not re.fullmatch(r"[a-z][a-z0-9_]{1,30}", source_id):
16 raise ValueError("identifiant : minuscules, chiffres et _ (2 à 31 caractères), commençant par une lettre")
17 target = SOURCES_FOLDER / f"{source_id}.py"
18 if target.exists():
19 raise FileExistsError(f"la source « {source_id} » existe déjà : {target}")
20 text = (SOURCES_FOLDER / "_template.py").read_text(encoding="utf-8")
21 for placeholder, value in {"__ID__": source_id, "__NAME__": name, "__COUNTRY_CODE__": country_code.upper(),
22 "__COUNTRY__": country, "__REGION__": region, "__FORMAT__": data_format,
23 "__LANG__": language.upper()}.items():
24 text = text.replace(placeholder, value)
25 target.write_text(text, encoding="utf-8")
26 return str(target.relative_to(config.ROOT))
