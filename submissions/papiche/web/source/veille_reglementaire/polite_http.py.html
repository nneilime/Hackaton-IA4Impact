2"""Client HTTP sobre et respectueux pour interroger les sources officielles.
3
4Objectif : faire le moins de requêtes possible, de façon honnête, pour ne jamais surcharger une source.
5- Un User-Agent qui dit qui nous sommes (aucun déguisement en navigateur).
6- Jeux de données ouverts téléchargés EN BLOC (quelques requêtes), jamais produit par produit.
7- Requêtes conditionnelles (ETag / Last-Modified) : si la source n'a pas changé, rien n'est retéléchargé (réponse 304).
8- Pause minimale entre deux requêtes vers le même serveur, et reprise polie quand un serveur demande d'attendre (429, 503).
9- Journal de CHAQUE requête sortante (hôte, adresse, statut, octets) : le nombre de requêtes se vérifie, il ne se promet pas.
10"""
11from __future__ import annotations
12
13import json
14import time
15from collections import defaultdict
16from datetime import datetime, timezone
17from pathlib import Path
18from urllib.parse import urlparse
19
20import requests
21
22from . import config
23
24USER_AGENT = "VeilleReglementairePPP/1.0 (hackathon; collecte respectueuse de données ouvertes)"
25MIN_SECONDS_BETWEEN_REQUESTS_TO_SAME_HOST = 2.0
26MAX_TRIES = 4
27RETRYABLE_STATUS = (429, 503)
28REQUEST_LOG = config.RAW / "_requests.jsonl"
29
30_session = requests.Session()
31_session.headers["User-Agent"] = USER_AGENT
32_last_request_time_by_host: dict[str, float] = {}
33
34
35# ── Journal des requêtes ─────────────────────────────────────────────────────
36
37def log_request(url: str, status: int, size_bytes: int, seconds: float) -> None:
38 config.RAW.mkdir(parents=True, exist_ok=True)
39 entry = {"at": datetime.now(timezone.utc).isoformat(timespec="seconds"), "host": urlparse(url).netloc,
40 "url": url, "status": status, "bytes": size_bytes, "seconds": round(seconds, 2)}
41 with REQUEST_LOG.open("a", encoding="utf-8") as file:
42 file.write(json.dumps(entry, ensure_ascii=False) + "\n")
43
44
45def read_request_log(since: str | None = None) -> list[dict]:
46 if not REQUEST_LOG.exists():
47 return []
48 entries = [json.loads(line) for line in REQUEST_LOG.read_text(encoding="utf-8").splitlines() if line.strip()]
49 return [e for e in entries if since is None or e["at"] >= since]
50
51
52def summarize_requests(entries: list[dict]) -> dict:
53 """Par serveur : nombre de requêtes, octets reçus, réponses « inchangé » (304)."""
54 per_host: dict[str, dict] = defaultdict(lambda: {"requests": 0, "megabytes": 0.0, "not_modified": 0})
55 for entry in entries:
56 host = per_host[entry["host"]]
57 host["requests"] += 1
58 host["megabytes"] += entry["bytes"] / 1e6
59 host["not_modified"] += entry["status"] == 304
60 for host in per_host.values():
61 host["megabytes"] = round(host["megabytes"], 1)
62 return {"total_requests": len(entries), "total_megabytes": round(sum(e["bytes"] for e in entries) / 1e6, 1),
63 "not_modified": sum(e["status"] == 304 for e in entries), "per_host": dict(per_host)}
64

66# ── Requêtes ─────────────────────────────────────────────────────────────────

68def wait_turn(host: str) -> None:
69 """Respecte une pause minimale entre deux requêtes vers le même serveur."""
70 elapsed = time.time() - _last_request_time_by_host.get(host, 0)
71 if elapsed < MIN_SECONDS_BETWEEN_REQUESTS_TO_SAME_HOST:
72 time.sleep(MIN_SECONDS_BETWEEN_REQUESTS_TO_SAME_HOST - elapsed)
73
74
75def seconds_to_wait(response: requests.Response, attempt: int) -> float:
76 """Le serveur dit combien attendre (Retry-After) ; sinon on double l'attente à chaque essai."""
77 retry_after = response.headers.get("Retry-After", "")
78 return float(retry_after) if retry_after.isdigit() else 5.0 * 2 ** attempt
79
80
81def send_request(url: str, headers: dict | None = None, stream: bool = False) -> requests.Response:
82 host = urlparse(url).netloc
83 for attempt in range(MAX_TRIES):
84 wait_turn(host)
85 started = time.time()
86 response = _session.get(url, headers=headers or {}, stream=stream, timeout=120)
87 _last_request_time_by_host[host] = time.time()
88 body_will_be_streamed = stream and response.status_code == 200
89 if not body_will_be_streamed: # un corps diffusé est journalisé par `download`, avec sa vraie taille
90 log_request(url, response.status_code, 0 if stream else len(response.content), time.time() - started)
91 if response.status_code not in RETRYABLE_STATUS:
92 response.started_at = started
93 return response
94 time.sleep(seconds_to_wait(response, attempt))
95 response.raise_for_status()
96 return response
97
98
99def get_json(url: str) -> dict:
100 response = send_request(url)
101 response.raise_for_status()
102 return response.json()

105def download(url: str, destination: Path) -> dict:
106 """Télécharge `url` vers `destination` seulement si la source a changé depuis la dernière fois.
107
108 Retourne {"changed": bool, "bytes": int}. Les validateurs (ETag, Last-Modified) sont gardés à côté du fichier.
109 """
110 destination.parent.mkdir(parents=True, exist_ok=True)
111 validators_file = destination.with_name(destination.name + ".validators.json")
112 headers = {}
113 if destination.exists() and validators_file.exists():
114 saved = json.loads(validators_file.read_text())
115 if saved.get("etag"):
116 headers["If-None-Match"] = saved["etag"]
117 if saved.get("last_modified"):
118 headers["If-Modified-Since"] = saved["last_modified"]
119
120 response = send_request(url, headers, stream=True)
121 if response.status_code == 304:
122 return {"changed": False, "bytes": 0}
123 response.raise_for_status()
124
125 partial_file = destination.with_name(destination.name + ".part") # écriture atomique
126 with partial_file.open("wb") as file:
127 for chunk in response.iter_content(1 << 20):
128 file.write(chunk)
129 partial_file.rename(destination)
130 log_request(url, response.status_code, destination.stat().st_size, time.time() - response.started_at)
131 validators_file.write_text(json.dumps({"etag": response.headers.get("ETag"),
132 "last_modified": response.headers.get("Last-Modified")}))
133 return {"changed": True, "bytes": destination.stat().st_size}
