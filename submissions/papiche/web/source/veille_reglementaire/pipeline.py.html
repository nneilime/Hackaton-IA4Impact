16def collect(only: list[str] | None = None) -> dict:
17 """Collecte les sources et garde la preuve de sobriété : combien de requêtes, vers quels serveurs."""
18 started_at = datetime.now(timezone.utc).isoformat(timespec="seconds")
19 results = []
20 for name, module in SOURCES.items():
21 if only and name not in only:
22 continue
23 print(f"[collecte] {name} ...", flush=True)
24 results.append(module.collect(scope()))
25 network = polite_http.summarize_requests(polite_http.read_request_log(since=started_at))
26 config.PROCESSED.mkdir(parents=True, exist_ok=True)
27 with (config.PROCESSED / "collect_runs.jsonl").open("a", encoding="utf-8") as file:
28 file.write(json.dumps({"at": started_at, "network": network}, ensure_ascii=False) + "\n")
29 print(f"[collecte] {network['total_requests']} requêtes, {network['total_megabytes']} Mo reçus, "
30 f"{network['not_modified']} réponse(s) « inchangé » (rien à retélécharger pour ces fichiers)", flush=True)
31 return {"results": results, "network": network}

34def index(only: list[str] | None = None) -> int:
35 if not only:
36 qdrant_store.reset()
37 n = 0
38 for name, mod in SOURCES.items():
39 if only and name not in only:
40 continue
41 try:
42 ps = mod.passages(scope())
43 except FileNotFoundError:
44 print(f"[index] {name}: données brutes absentes (lancer `collect`)", file=sys.stderr)
45 continue
46 k = qdrant_store.index_passages(ps)
47 print(f"[index] {name}: {k} passages -> Qdrant", flush=True)
48 n += k
49 return n

52def extract(limit: int | None = None, product: str | None = None, source: str | None = None,
53 per_product: int | None = None) -> dict:
54 passages = qdrant_store.all_passages(**({"source_id": source} if source else {}))
55 if product:
56 passages = [p for p in passages if product.casefold() in p.product.casefold()]
57 if per_product:
58 seen, keep = {}, []
59 for p in passages:
60 key = (p.product, p.country)
61 if seen.get(key, 0) < per_product:
62 seen[key] = seen.get(key, 0) + 1
63 keep.append(p)
64 passages = keep
65 if limit:
66 passages = passages[:limit]
67 traces, records, hitl = [], [], []
68 t0 = time.time()
69 for i, p in enumerate(passages, 1):
70 tr = extract_one(p)
71 traces.append(tr)
72 (records if tr["outcome"] == "accepted" else hitl).append(tr["record"] or {"passage_id": p.passage_id})
73 print(f"[{i}/{len(passages)}] {p.country} {p.product[:22]:22} {tr['outcome']:8} essais={len(tr['attempts'])} "
74 f"({time.time() - t0:.0f}s)", flush=True)
75 config.PROCESSED.mkdir(parents=True, exist_ok=True)
76 _dump("traces.jsonl", traces)
77 _dump("records.jsonl", records)
78 _dump("hitl.jsonl", hitl)
79 qdrant_store.upsert_records(records)
80 meta = [{"product": p.product, "country": p.country, "ai": p.ai, "holder": p.holder} for p in qdrant_store.all_passages()]
81 _dump("sisters.jsonl", sister_products.detect(meta))
82 from . import review # import tardif : review utilise pipeline
83 review.apply_stored_corrections()
84 return {"passages": len(passages), "records": len(records), "hitl": len(hitl)}
