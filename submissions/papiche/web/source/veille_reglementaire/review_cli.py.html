10def resolve(target: str, queue: list[dict]) -> str:
11 """« 3 » = troisième ligne de `./veille review` ; sinon l'identifiant complet du passage."""
12 if target.isdigit():
13 index = int(target) - 1
14 if not 0 <= index < len(queue):
15 raise SystemExit(f"Numéro hors liste (1 à {len(queue)}). Lancer `./veille review`.")
16 return queue[index]["passage_id"]
17 return target
18
19
20def print_queue(queue: list[dict]) -> None:
21 if not queue:
22 print("Rien à relire.")
23 return
24 print(f"{len(queue)} source(s) à relire :\n")
25 for number, item in enumerate(queue, 1):
26 label = "à corriger" if item["kind"] == "a_corriger" else "à confirmer"
27 print(f"{number:>3} {label:11} {item['country']} {item['product'][:26]:26} {item['reason'][:70]}")
28 print("\nVoir le détail : ./veille review --show N · écran de relecture : ./veille serve puis http://localhost:8088/review")
29
30
31def print_detail(item: dict) -> None:
32 print(f"{item['product']} ({item['country']}, {item['source_id']}) — {item['kind']}")
33 print(f"Source officielle : {item['source_url']} (collectée le {item['collected_at']})")
34 print(f"\nPourquoi : {item['reason']}")
35 print(f"\n--- texte source ---\n{item['text']}\n")
36 print("--- dernière proposition de l'IA ---\n" + json.dumps(item["proposal"], ensure_ascii=False, indent=1))
37
38
39def run(args) -> None:
40 if args.cmd == "review":
41 queue = review.review_queue(include_decided=args.all)
42 if args.show:
43 print_detail(next(i for i in queue if i["passage_id"] == resolve(args.show, queue)))
44 else:
45 print_queue(queue)
46 return
47 queue = review.review_queue()
48 passage_id = resolve(args.target, queue)
49 try:
50 if args.cmd == "confirm":
51 review.decide(passage_id, "confirmed", args.author)
52 elif args.cmd == "dismiss":
53 review.decide(passage_id, "dismissed", args.author, reason=args.reason)
54 else:
55 item = next(i for i in review.review_queue(include_decided=True) if i["passage_id"] == passage_id)
56 fields = dict(item["proposal"]) # on repart de la proposition de l'IA…
57 for name in review.EXTRACTION_FIELDS: # …et les options de l'expert la remplacent
58 value = getattr(args, name, None)
59 if value is not None:
60 fields[name] = value
61 review.decide(passage_id, "corrected", args.author, fields=fields)
62 except ValueError as error:
63 raise SystemExit(f"Refusé : {error}")
64 print("Décision enregistrée.")
