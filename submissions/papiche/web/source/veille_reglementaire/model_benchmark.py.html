2"""Comparatif de modèles d'IA sur la MÊME chaîne d'extraction, pour dimensionner GPU et stockage.
3
4MESURÉ (sur le service d'IA de l'essaim) : qualité (exactitude vs source structurée), essais, latence,
5tokens/s, taille disque et mémoire GPU du modèle chargé (API Ollama).
6ESTIMÉ (toujours étiqueté comme tel) : VRAM d'un autre contexte ou d'un autre nombre de requêtes parallèles,
7et débit sur d'autres cartes. L'efficacité utilisée pour ces estimations est CALIBRÉE sur la mesure réelle.
8Le script se rejoue tel quel sur la machine GPU cible : `./veille bench`.

30def pick_sample(passages, size_per_source: dict | None = None) -> list:
31 """Échantillon déterministe, identique pour tous les modèles : pas régulier dans chaque source."""
32 size_per_source = size_per_source or SAMPLE_SIZE_PER_SOURCE
33 sample = []
34 for source_id, wanted in size_per_source.items():
35 in_source = sorted((p for p in passages if p.source_id == source_id), key=lambda p: p.passage_id)
36 step = max(1, len(in_source) // wanted)
37 sample += in_source[::step][:wanted]
38 return sample

41def call_ollama_api(path: str, payload: dict | None = None) -> dict:
42 url = config.OLLAMA_HOST.rstrip("/") + path
43 response = requests.post(url, json=payload, timeout=60) if payload else requests.get(url, timeout=60)
44 response.raise_for_status()
45 return response.json()

48# ── Faits sur un modèle (taille, architecture, coût du cache KV) ─────────────

50def read_model_facts(model_name: str) -> dict:
51 installed = {m["name"]: m for m in call_ollama_api("/api/tags")["models"]}
52 if model_name not in installed and f"{model_name}:latest" in installed:
53 model_name = f"{model_name}:latest"
54 info = call_ollama_api("/api/show", {"model": model_name})
55 details, model_info = info["details"], info["model_info"]
56 architecture = model_info["general.architecture"]
57
58 def arch_value(key, default=None):
59 return model_info.get(f"{architecture}.{key}", default)
60
61 layers = arch_value("block_count")
62 heads = arch_value("attention.head_count")
63 kv_heads = arch_value("attention.head_count_kv", heads)
64 head_dim = arch_value("attention.key_length") or (arch_value("embedding_length") // heads if heads else None)
65 # Cache KV par token = clés + valeurs (×2), en demi-précision (2 octets). Borne haute (ignore les fenêtres glissantes).
66 kv_bytes_per_token = 2 * layers * kv_heads * head_dim * 2 if layers and kv_heads and head_dim else None
67 return {"model": model_name, "params": details.get("parameter_size"), "quant": details.get("quantization_level"),
68 "disk_gb": round(installed[model_name]["size"] / 1e9, 2) if model_name in installed else None,
69 "arch": architecture, "layers": layers, "kv_heads": kv_heads, "head_dim": head_dim,
70 "context_max": arch_value("context_length"), "kv_bytes_per_token": kv_bytes_per_token}

73def read_loaded_memory(model_name: str) -> dict:
74 """Mémoire réellement occupée par le modèle chargé (poids + cache KV), et part qui est en VRAM."""
75 for loaded in call_ollama_api("/api/ps").get("models", []):
76 if model_name in (loaded["name"], loaded["model"]):
77 return {"total_gb": round(loaded["size"] / 1e9, 2), "vram_gb": round(loaded.get("size_vram", 0) / 1e9, 2)}
78 return {"total_gb": None, "vram_gb": None}

81# ── Mesure d'un modèle ───────────────────────────────────────────────────────
82
83def summarize_by_source(traces: list[dict]) -> dict:
84 summary: dict[str, dict] = {}
85 for trace in traces:
86 source = summary.setdefault(trace["passage"]["source_id"], {"n": 0, "accepted": 0, "attempts": 0, "verifie_auto": 0})
87 source["n"] += 1
88 source["accepted"] += trace["outcome"] == "accepted"
89 source["attempts"] += len(trace["attempts"])
90 source["verifie_auto"] += bool(trace["record"]) and trace["record"]["validation_status"] == "verifie_auto"
91 for source in summary.values():
92 source["accept_rate"] = round(source["accepted"] / source["n"], 3)
93 source["mean_attempts"] = round(source["attempts"] / source["n"], 2)
94 return summary

97def measure_model(model_name: str, passages: list) -> dict:
98 """Passe tous les passages de l'échantillon dans la chaîne complète avec ce modèle (sans cache)."""
99 config.MODEL, llm.DIRECT, llm.BYPASS_CACHE = model_name, True, True
100 llm.COUNTERS.update({key: 0 for key in llm.COUNTERS})
101
102 traces, seconds_per_passage = [], []
103 for number, passage in enumerate(passages, 1):
104 started = time.time()
105 trace = extraction_loop.extract_passage(passage)
106 seconds_per_passage.append(time.time() - started)
107 traces.append(trace)
108 print(f" [{model_name}] {number}/{len(passages)} {passage.source_id} {trace['outcome']} "
109 f"essais={len(trace['attempts'])} {seconds_per_passage[-1]:.1f}s", flush=True)
110
111 counters = dict(llm.COUNTERS)
112 expected = {p.passage_id: p.gold for p in passages if p.gold}
113 quality = evaluation.score_against_source(traces, expected)
114 by_source = summarize_by_source(traces)
115 count = len(passages)
116 return {
117 "model": model_name, "passages": count,
118 "accepted": sum(s["accepted"] for s in by_source.values()), "hitl": count - sum(s["accepted"] for s in by_source.values()),
119 "verifie_auto": sum(s["verifie_auto"] for s in by_source.values()), "by_source": by_source,
120 "accuracy_vs_structured": quality["exactitude_par_champ"], "invented_values_published": quality["valeurs_inventees_publiees"],
121 "s_per_passage_mean": round(sum(seconds_per_passage) / count, 2),
122 "s_per_passage_p95": round(sorted(seconds_per_passage)[int(0.95 * (count - 1))], 2),
123 "llm_calls": counters["calls"], "prompt_tokens_per_passage": round(counters["prompt_tokens"] / count),
124 "gen_tokens_per_passage": round(counters["gen_tokens"] / count),
125 "decode_tok_s": round(counters["gen_tokens"] / counters["decode_s"], 1) if counters["decode_s"] else None,
126 "prefill_tok_s": round(counters["prompt_tokens"] / counters["prefill_s"], 1) if counters["prefill_s"] else None,
127 "load_s": round(counters["load_s"], 1), "memory_loaded": read_loaded_memory(model_name),
128 }

131# ── Estimations : VRAM, débit par carte, recommandation ──────────────────────
132
133def estimate_vram(facts: dict, parallel_requests: int = 1) -> dict:
134 """VRAM = poids + cache KV (contexte × requêtes parallèles) + marge. ESTIMATION."""
135 kv_gb = (facts["kv_bytes_per_token"] or 0) * CONTEXT_TOKENS * parallel_requests / 1e9
136 return {"parallel": parallel_requests, "weights_gb": facts["disk_gb"], "kv_gb": round(kv_gb, 2),
137 "vram_gb": round(facts["disk_gb"] + kv_gb + GPU_OVERHEAD_GB, 2)}

144def find_declared_gpu() -> dict | None:
145 """La carte du service d'IA est déclarée (VEILLE_GPU) : l'API Ollama ne donne pas son nom."""
146 declared = config.SERVICE_GPU.casefold()
147 return next((g for g in load_gpu_catalog() if declared and declared in g["name"].casefold()), None)

150def calibrate_efficiency(results: dict, facts_by_model: dict, gpu: dict | None) -> float | None:
151 """Part de la bande passante de la carte réellement utilisée en génération, mesurée sur le plus gros modèle.
152
153 Débit effectif (Go/s) = tokens/s mesurés × taille des poids. Efficacité = débit effectif / bande passante de la fiche.
154 """
155 if not gpu:
156 return None
157 biggest = max(results, key=lambda m: facts_by_model[m]["disk_gb"])
158 effective_gb_s = results[biggest]["decode_tok_s"] * facts_by_model[biggest]["disk_gb"]
159 return round(effective_gb_s / gpu["bandwidth_gb_s"], 2)
160
161
162def estimate_gpu_table(results: dict, facts_by_model: dict, efficiency: float | None) -> list[dict]:
163 efficiency = efficiency or 0.6 # valeur prudente si aucune carte n'est déclarée
164 rows = []
165 for gpu in load_gpu_catalog():
166 for model, facts in facts_by_model.items():
167 vram_needed = estimate_vram(facts)["vram_gb"]
168 fits = vram_needed <= gpu["vram_gb"]
169 tokens_per_second = efficiency * gpu["bandwidth_gb_s"] / facts["disk_gb"]
170 seconds_per_passage = results[model]["gen_tokens_per_passage"] / tokens_per_second
171 rows.append({"gpu": gpu["name"], "model": model, "fits": fits, "vram_needed_gb": vram_needed,
172 "est_tok_s": round(tokens_per_second),
173 "est_passages_per_hour": round(3600 / seconds_per_passage) if fits else None})
174 return rows

177def recommend(results: dict, facts_by_model: dict) -> dict:
178 """Le plus petit modèle dont la qualité est à moins de ACCEPTED_TOLERANCE du meilleur."""
179 best_accepted = max(r["accepted"] for r in results.values())
180 good_enough = [m for m, r in results.items() if r["accepted"] >= best_accepted - ACCEPTED_TOLERANCE]
181 chosen = min(good_enough, key=lambda m: facts_by_model[m]["disk_gb"])
182 fitting_gpus = [g for g in load_gpu_catalog() if estimate_vram(facts_by_model[chosen], 4)["vram_gb"] <= g["vram_gb"]]
183 return {"model": chosen, "vram_1_request_gb": estimate_vram(facts_by_model[chosen], 1)["vram_gb"],
184 "vram_4_parallel_gb": estimate_vram(facts_by_model[chosen], 4)["vram_gb"],
185 "smallest_gpu": min(fitting_gpus, key=lambda g: g["vram_gb"])["name"] if fitting_gpus else None,
186 "rule": f"plus petit modèle à moins de {ACCEPTED_TOLERANCE} passages du meilleur ({best_accepted} acceptés)"}

191def folder_size_mb(folder) -> float:
192 if not folder.exists():
193 return 0.0
194 return round(sum(f.stat().st_size for f in folder.rglob("*") if f.is_file()) / 1e6, 1)
195
196
197def measure_storage() -> dict:
198 """Volumes réels de ce dépôt, puis extrapolation linéaire (hypothèses écrites dans la note)."""
199 raw_mb = {source: folder_size_mb(config.RAW / source) for source in ("ephy", "pmra", "epa")}
200 passages_now = len(qdrant_store.all_passages())
201 vector_dimensions = 768
202 kb_per_passage = (vector_dimensions * 4 + 2200) / 1024 # vecteur f32 + charge utile typique (~2 Ko), hors index
203 index_margin = 1.6 # index HNSW et réplication
204 scenarios = [("MVP actuel (3 sources, produits LPC)", passages_now),
205 ("Catalogues complets FR + CA + US", 400_000), ("Monde (≈ 10 pays)", 1_000_000)]
206 return {"raw_mb": raw_mb, "raw_total_mb": round(sum(raw_mb.values()), 1),
207 "processed_mb": folder_size_mb(config.PROCESSED), "llm_cache_mb": folder_size_mb(config.CACHE),
208 "passages_now": passages_now, "kb_per_passage": round(kb_per_passage, 1),
209 "extrapolation": [{"scenario": name, "passages": n, "qdrant_gb_est": round(n * kb_per_passage * index_margin / 1e6, 2)}
210 for name, n in scenarios],
211 "note": "Extrapolation linéaire (×1,6 pour l'index et la réplication), à recalibrer sur le volume réel."}

216def describe_service(results: dict, gpu: dict | None, efficiency: float | None) -> dict:
217 on_gpu = any((r["memory_loaded"]["vram_gb"] or 0) > 0 for r in results.values())
218 client_cpu = next((l.split(":", 1)[1].strip() for l in open("/proc/cpuinfo") if l.startswith("model name")), "")
219 return {"accelerator": "GPU" if on_gpu else "CPU", "gpu_declared": gpu["name"] if gpu else (config.SERVICE_GPU or None),
220 "gpu_bandwidth_gb_s": gpu["bandwidth_gb_s"] if gpu else None, "efficiency_measured": efficiency,
221 "endpoint": config.OLLAMA_HOST,
222 "client": {"cpu": client_cpu, "threads": os.cpu_count(), "has_local_nvidia": bool(shutil.which("nvidia-smi")),
223 "note": "le poste client n'exécute pas le modèle : le calcul se fait sur le service d'IA de l'essaim"}}

226def write_report(results: dict, facts_by_model: dict, sample_passages: list) -> dict:
227 gpu = find_declared_gpu()
228 efficiency = calibrate_efficiency(results, facts_by_model, gpu)
229 for model, result in results.items():
230 result["facts"] = facts_by_model[model]
231 result["sizing"] = {"1 requête": estimate_vram(facts_by_model[model], 1),
232 "4 requêtes parallèles": estimate_vram(facts_by_model[model], 4)}
233 report = {
234 "service": describe_service(results, gpu, efficiency),
235 "sample": {"passages": len(sample_passages),
236 "by_source": {s: sum(p.source_id == s for p in sample_passages) for s in SAMPLE_SIZE_PER_SOURCE}},
237 "results": results, "recommendation": recommend(results, facts_by_model),
238 "gpu_table": estimate_gpu_table(results, facts_by_model, efficiency),
239 "storage": measure_storage(), "embedding": read_model_facts(config.EMBED_MODEL)}
240 report_dir = config.PROCESSED / "bench"
241 report_dir.mkdir(exist_ok=True)
242 (report_dir / "report.json").write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
243 (config.ROOT / "docs").mkdir(exist_ok=True)
244 (config.ROOT / "docs" / "BENCHMARK.md").write_text(render_markdown(report), encoding="utf-8")
245 return report

248def run(models: list[str] | None = None, size_per_source: dict | None = None) -> dict:
249 models = models or DEFAULT_MODELS
250 sample_passages = pick_sample(qdrant_store.all_passages(), size_per_source)
251 print(f"[bench] {len(sample_passages)} passages, modèles : {', '.join(models)}", flush=True)
252 results, facts_by_model = {}, {}
253 try:
254 for model in models:
255 facts_by_model[model] = read_model_facts(model)
256 results[model] = measure_model(model, sample_passages)
257 finally:
258 llm.DIRECT = llm.BYPASS_CACHE = False
259 return write_report(results, facts_by_model, sample_passages)
