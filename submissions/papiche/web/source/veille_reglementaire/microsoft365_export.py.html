2"""Exports pour l'écosystème Microsoft 365 : contrat OpenAPI (connecteur Power Platform / Copilot Studio),
3schéma de table Dataverse et CSV comparatif (import SharePoint / Excel)."""
4from __future__ import annotations
5import csv, json
6from . import comparison_table, config, pipeline
7from .models import Record
8
9OUT = config.ROOT / "m365"
10
11DATAVERSE_TYPES = {"string": "Texte (une ligne)", "number": "Nombre décimal", "integer": "Nombre entier", "array": "Texte multiligne (JSON)",
12 "object": "Texte multiligne (JSON)", "boolean": "Oui/Non"}
13
14
15def export():
16 OUT.mkdir(exist_ok=True)
17 from .api import app
18 (OUT / "openapi.json").write_text(json.dumps(app.openapi(), ensure_ascii=False, indent=1), encoding="utf-8")
19 rows = comparison_table.table(pipeline.load("records.jsonl"))
20 (OUT / "comparatif_produit_pays_usage.csv").write_text(comparison_table.to_csv(rows), encoding="utf-8-sig") # BOM : Excel FR
21 props = Record.model_json_schema()["properties"]
22 with open(OUT / "dataverse_table_lpc_record.csv", "w", encoding="utf-8", newline="") as f:
23 w = csv.writer(f)
24 w.writerow(["Nom logique", "Type Dataverse suggéré", "Description"])
25 for k, v in props.items():
26 t = v.get("type") or next((a.get("type") for a in v.get("anyOf", []) if a.get("type") != "null"), "string")
27 w.writerow([f"lpc_{k}", DATAVERSE_TYPES.get(t, "Texte (une ligne)"), v.get("title", k)])
28 print(f"m365/ : openapi.json, comparatif ({len(rows)} lignes), dataverse_table_lpc_record.csv")
