7COLS = ["product", "country", "authorization_status", "crop_original", "crop_eppo", "target_original", "target_eppo",
8 "dose_original", "dose_norm_value", "dose_norm_unit", "max_applications", "pre_harvest_interval_days",
9 "validation_status", "source_id", "source_url", "collected_at", "quote"]

12def flat(r: dict) -> dict:
13 c = r["citation"]
14 d = {k: r.get(k) for k in COLS if k not in ("source_id", "source_url", "collected_at", "quote")}
15 d.update(source_id=c["source_id"], source_url=c["source_url"], collected_at=c["collected_at"], quote=c["quote"])
16 return d

19def table(records: list[dict], product: str | None = None, countries: list[str] | None = None,
20 crop: str | None = None, target: str | None = None) -> list[dict]:
21 def keep(r):
22 if product and product.casefold() not in r["product"].casefold():
23 return False
24 if countries and r["country"] not in countries:
25 return False
26 if crop and crop.casefold() not in f"{r.get('crop_original') or ''} {r.get('crop_eppo') or ''}".casefold():
27 return False
28 if target and target.casefold() not in f"{r.get('target_original') or ''} {r.get('target_eppo') or ''}".casefold():
29 return False
30 return True
31 rows = [flat(r) for r in records if keep(r)]
32 return sorted(rows, key=lambda x: (x["product"], x["country"], x["crop_original"] or "", x["target_original"] or ""))

35def footprint(records: list[dict]) -> list[dict]:
36 """Empreinte réglementaire : par produit et pays, statut et nombre d'usages documentés."""
37 agg = defaultdict(lambda: {"usages": 0, "with_dose": 0, "status": set(), "validated": 0})
38 for r in records:
39 a = agg[(r["product"], r["country"])]
40 a["usages"] += 1
41 a["with_dose"] += r.get("dose_value") is not None
42 a["status"].add(r.get("authorization_status_original") or "?")
43 a["validated"] += r["validation_status"] == "verifie_auto"
44 return [{"product": p, "country": c, "usages": a["usages"], "with_dose": a["with_dose"],
45 "statuses": sorted(a["status"]), "verifie_auto": a["validated"]}
46 for (p, c), a in sorted(agg.items())]

49def to_csv(rows: list[dict]) -> str:
50 buf = io.StringIO()
51 w = csv.DictWriter(buf, fieldnames=COLS, extrasaction="ignore")
52 w.writeheader()
53 w.writerows(rows)
54 return buf.getvalue()
