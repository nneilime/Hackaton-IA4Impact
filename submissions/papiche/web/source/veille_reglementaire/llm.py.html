2"""Adaptateur LLM : délègue à question.py (Astroport.ONE/IA) s'il est présent, sinon Ollama direct.
3
4Le fournisseur est interchangeable par configuration (VEILLE_MODEL, OLLAMA_HOST) : aucune clé requise.
5Chaque appel est mis en cache (hash du prompt) pour qu'un rejeu soit instantané et reproductible.
6"""
7from __future__ import annotations
8import hashlib, json, sys
9from pathlib import Path
10from . import config
11
12_question = None
13_tried = False
14DIRECT = False # benchmark : client Ollama direct (question.py ne remonte pas les compteurs de tokens)
15BYPASS_CACHE = False # benchmark : mesurer le vrai calcul, jamais une réponse en cache
16COUNTERS = {"calls": 0, "prompt_tokens": 0, "gen_tokens": 0, "prefill_s": 0.0, "decode_s": 0.0, "load_s": 0.0, "total_s": 0.0}
17

19def _load_question_py():
20 """Importe question.py sans effet de bord (son execv n'est actif qu'en exécution directe)."""
21 global _question, _tried
22 if _tried:
23 return _question
24 _tried = True
25 d = Path(config.QUESTION_PY_DIR)
26 if (d / "question.py").is_file():
27 sys.path.insert(0, str(d))
28 try:
29 import question # type: ignore
30 _question = question
31 except Exception as e: # dépendances absentes, etc. -> repli propre
32 print(f"[llm] question.py indisponible ({e}); repli Ollama direct", file=sys.stderr)
33 return _question
34
35
36def backend() -> str:
37 return "question.py" if _load_question_py() else "ollama"
38

40def _key(model: str, system: str, user: str) -> str:
41 return hashlib.sha256(f"{model}\x00{system}\x00{user}".encode()).hexdigest()
42
43
44def chat_json(system: str, user: str, *, model: str | None = None, temperature: float = 0.0,
45 num_ctx: int = 4096, num_predict: int = 700) -> dict:
46 """Retourne {"text": str|None, "cached": bool, "backend": str, "model": str}."""
47 model = model or config.MODEL
48 path = config.CACHE / f"{_key(model, system, user)}.json"
49 if path.is_file() and not BYPASS_CACHE:
50 d = json.loads(path.read_text())
51 d["cached"] = True
52 return d
53 q = None if DIRECT else _load_question_py()
54 text, stats = None, None
55 if q is not None:
56 text = q.get_ollama_answer(user, model, system, temperature=temperature, num_ctx=num_ctx,
57 num_predict=num_predict, format_json=True)
58 else:
59 import ollama
60 client = ollama.Client(host=config.OLLAMA_HOST)
61 r = client.chat(model=model, format="json",
62 messages=[{"role": "system", "content": system}, {"role": "user", "content": user}],
63 options={"temperature": temperature, "num_ctx": num_ctx, "num_predict": num_predict})
64 text = r["message"]["content"]
65 ns = 1e9
66 stats = {"prompt_tokens": r.get("prompt_eval_count") or 0, "gen_tokens": r.get("eval_count") or 0,
67 "prefill_s": (r.get("prompt_eval_duration") or 0) / ns, "decode_s": (r.get("eval_duration") or 0) / ns,
68 "load_s": (r.get("load_duration") or 0) / ns, "total_s": (r.get("total_duration") or 0) / ns}
69 COUNTERS["calls"] += 1
70 for k, v in stats.items():
71 COUNTERS[k] += v
72 out = {"text": text, "cached": False, "backend": "ollama" if q is None else backend(), "model": model, "stats": stats}
73 if text is not None and not BYPASS_CACHE:
74 path.write_text(json.dumps(out, ensure_ascii=False))
75 return out
