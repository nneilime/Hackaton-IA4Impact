2"""Une nouvelle source se branche sans toucher au reste du code : découverte automatique et squelette valide.
3
4Lancer : python -m unittest discover -s tests -v (aucun service externe requis)
5"""
6import sys
7import unittest
8from pathlib import Path
9
10sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
11
12from veille_reglementaire import source_scaffold
13from veille_reglementaire.sources import discover_sources
14
15
16class Sources(unittest.TestCase):
17 def test_les_trois_sources_actuelles_sont_decouvertes(self):
18 self.assertTrue({"ephy", "pmra", "epa"} <= set(discover_sources()))
19
20 def test_le_modele_n_est_pas_une_source(self):
21 self.assertNotIn("__ID__", discover_sources())
22
23 def test_une_nouvelle_source_est_decouverte_et_valide(self):
24 created = Path(source_scaffold.create_source("essai_test", "Essai", "ZZ", "Pays d'essai", "Région", "CSV", "fr"))
25 try:
26 sources = discover_sources()
27 self.assertIn("essai_test", sources)
28 module = sources["essai_test"]
29 self.assertEqual(module.INFO["country_code"], "ZZ")
30 self.assertEqual(module.passages({"products": []}), [])
31 with self.assertRaises(FileExistsError):
32 source_scaffold.create_source("essai_test", "Essai", "ZZ", "Pays", "Région")
33 finally:
34 (source_scaffold.config.ROOT / created).unlink(missing_ok=True)
35
36 def test_identifiant_invalide_refuse(self):
37 with self.assertRaises(ValueError):
38 source_scaffold.create_source("Mauvais Nom", "x", "ZZ", "x", "x")
39
40
41if __name__ == "__main__":
42 unittest.main()
