16TEXT = ("Produit: LALSTOP G46 WG (AMM 2230771) | Usage: Vigne*Trt Part.Aer.*Pourriture grise | "

21TRACE = {"passage": {"passage_id": "t:1"}, "attempts": [{"n": 1, "model": "fake", "raw": "{}", "error": "preuve non retrouvée"}],
22 "outcome": "hitl", "record": None, "critic": None}
23GOOD = {"crop_original": "Vigne", "target_original": "Pourriture grise", "dose_original": "0.5 kg/ha", "dose_value": 0.5,

27class Relecture(unittest.TestCase):
28 def setUp(self):
29 self.get_passage, review.qdrant_store.get_passage = review.qdrant_store.get_passage, lambda _id: PASSAGE
30
31 def tearDown(self):
32 review.qdrant_store.get_passage = self.get_passage
33
34 def test_correction_valide_donne_le_statut_humain(self):
35 record = review.build_human_record(TRACE, GOOD, "Camille")
36 self.assertEqual(record["validation_status"], "valide_humain")
37 self.assertEqual(record["dose_value"], 0.5)
38 self.assertIn("Camille", record["model"])
39
40 def test_une_dose_inventee_est_refusee_meme_par_un_humain(self):
41 fabricated = dict(GOOD, dose_original="1.2 kg/ha", dose_value=1.2, quote="Dose retenue: 1.2 kg/ha")
42 with self.assertRaisesRegex(ValueError, "preuve non retrouvée"):
43 review.build_human_record(TRACE, fabricated, "Camille")
44
45 def test_champ_inconnu_refuse(self):
46 with self.assertRaises(ValueError):
47 review.build_human_record(TRACE, dict(GOOD, invente=True), "Camille")
48
49 def test_ecarter_exige_une_raison(self):
50 with self.assertRaisesRegex(ValueError, "raison"):
51 review.decide("t:1", "dismissed", "Camille", reason=" ")
52
53 def test_decision_enregistree_puis_relue(self):
54 with tempfile.TemporaryDirectory() as folder:
55 old = (review.CORRECTIONS_FILE, review.apply_decision, review.config.PROCESSED)
56 review.CORRECTIONS_FILE = Path(folder) / "corrections.jsonl"
57 review.apply_decision = lambda entry: None # l'écriture des enregistrements est testée ailleurs
58 try:
59 review.decide("t:1", "dismissed", "Camille", reason="étiquette sans dose")
60 self.assertEqual(review.load_decisions()["t:1"]["action"], "dismissed")
61 review.decide("t:1", "dismissed", "Léo", reason="doublon") # la dernière décision l'emporte
62 self.assertEqual(review.load_decisions()["t:1"]["author"], "Léo")
63 finally:
64 review.CORRECTIONS_FILE, review.apply_decision, _ = old
