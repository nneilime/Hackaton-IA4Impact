138 function planRoute(area) {
139 const n = i => node(area, i), dip = (from, to) => ({ x: (n(from).x + n(to).x) / 2, y: n(from).y + area.h * 0.24 });
140 const roll = Math.random();
141 if (roll < share.pass) return { points: [0, 1, 2, 3, 4, 5].map(n), kind: 'pass' };
142 if (roll < share.pass + share.retry) return { points: [n(0), n(1), n(2), n(3), dip(3, 2), n(2), n(3), n(4), n(5)], kind: 'retry' };
143 return { points: [n(0), n(1), n(2), n(3), dip(3, 2), n(2), n(3), dip(3, 2), n(2), n(3), expertNode(area)], kind: 'expert' };
144 }

160 lastSpawn = t; const route = planRoute(area);
161 particles.push({ ...route, index: 1, x: route.points[0].x, y: route.points[0].y, color: COLOR.blue, failed: 0 });
162 }
163 particles = particles.filter(particle => {
164 const target = particle.points[particle.index];
165 const dx = target.x - particle.x, dy = target.y - particle.y, distance = Math.hypot(dx, dy), step = 5.2 * u;

173 }

175 if (particle.kind === 'pass' && particle.index > 3) particle.color = COLOR.green;

180 });
